16QVector<double> PolyTransform::basis(double x, double y, int order)
17{
18 QVector<double> t;
19 for (int i = 0; i <= order; ++i)
20 for (int j = 0; j <= i; ++j)
21 t.append(std::pow(x, i - j) * std::pow(y, j));
22 return t;
23 // order=1: [1, x, y]
24 // order=2: [1, x, y, x², xy, y²]
25 // order=3: [1, x, y, x², xy, y², x³, x²y, xy², y³]
26}

28bool PolyTransform::gaussElim(QVector<QVector<double>> A, QVector<double> &b)
29{
30 const int n = A.size();
31 for (int col = 0; col < n; ++col)
32 {
33 // Partial pivot
34 int piv = col;
35 for (int r = col + 1; r < n; ++r)
36 if (std::abs(A[r][col]) > std::abs(A[piv][col])) piv = r;
37 std::swap(A[col], A[piv]);
38 std::swap(b[col], b[piv]);
39
40 if (std::abs(A[col][col]) < 1e-14) return false;
41
42 for (int r = col + 1; r < n; ++r)
43 {
44 const double f = A[r][col] / A[col][col];
45 for (int k = col; k < n; ++k) A[r][k] -= f * A[col][k];
46 b[r] -= f * b[col];
47 }
48 }
49 for (int r = n - 1; r >= 0; --r)
50 {
51 for (int c = r + 1; c < n; ++c) b[r] -= A[r][c] * b[c];
52 b[r] /= A[r][r];
53 }
54 return true;
55}

57bool PolyTransform::fit(const QVector<GcpPoint> &gcps, int order)
58{
59 mValid = false;
60 mOrder = order;
61 const int nT = (order + 1) * (order + 2) / 2;
62
63 // Count active GCPs
64 int nA = 0;
65 for (const auto &g : gcps) if (g.enabled) ++nA;
66 if (nA < nT) return false;
67
68 // Build normal equations A^T A x = A^T b (once for E, once for N)
69 QVector<QVector<double>> ATA(nT, QVector<double>(nT, 0.0));
70 QVector<double> ATbE(nT, 0.0), ATbN(nT, 0.0);
71
72 for (const auto &g : gcps)
73 {
74 if (!g.enabled) continue;
75 const auto bs = basis(g.srcX, g.srcY, order);
76 for (int i = 0; i < nT; ++i)
77 {
78 for (int j = 0; j < nT; ++j) ATA[i][j] += bs[i] * bs[j];
79 ATbE[i] += bs[i] * g.refX;
80 ATbN[i] += bs[i] * g.refY;
81 }
82 }
83
84 auto ATA2 = ATA;
85 mCoeffE = ATbE;
86 mCoeffN = ATbN;
87
88 if (!gaussElim(ATA, mCoeffE) || !gaussElim(ATA2, mCoeffN))
89 return false;
90
91 mValid = true;
92 return true;
93}

95void PolyTransform::transform(double x, double y, double &e, double &n) const
96{
97 e = n = 0;
98 if (!mValid) return;
99 const auto bs = basis(x, y, mOrder);
100 for (int i = 0; i < mCoeffE.size(); ++i) e += mCoeffE[i] * bs[i];
101 for (int i = 0; i < mCoeffN.size(); ++i) n += mCoeffN[i] * bs[i];
102}

104void PolyTransform::computeResiduals(QVector<GcpPoint> &gcps) const
105{
106 for (auto &g : gcps)
107 {
108 if (!mValid || !g.enabled) { g.resX = g.resY = g.rms = 0; continue; }
109 double predE, predN;
110 transform(g.srcX, g.srcY, predE, predN);
111 g.resX = predE - g.refX;
112 g.resY = predN - g.refY;
113 g.rms = std::sqrt(g.resX * g.resX + g.resY * g.resY);
114 }
115}

117double PolyTransform::totalRms(const QVector<GcpPoint> &gcps) const
118{
119 double sum = 0; int n = 0;
120 for (const auto &g : gcps)
121 if (g.enabled && g.rms > 0) { sum += g.rms * g.rms; ++n; }
122 return n > 0 ? std::sqrt(sum / n) : 0.0;
123}

236QPointF GrassGcpCanvas::mapToWidget(double east, double north) const
237{
238 if (mPixmap.isNull()) return {};
239 const double pw = mPixmap.width();
240 const double ph = mPixmap.height();
241 const double ex = (east - mExtW) / (mExtE - mExtW);
242 const double ey = (mExtN - north) / (mExtN - mExtS);
243 return { mOffset.x() + ex * pw * mScale,
244 mOffset.y() + ey * ph * mScale };

248{
