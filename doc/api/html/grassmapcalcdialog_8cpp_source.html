29// ============================================================
30class GrassMapCalcHighlighter : public QSyntaxHighlighter
31{
32 public:
33 explicit GrassMapCalcHighlighter( QTextDocument *doc )
34 : QSyntaxHighlighter( doc )
35 {
36 // Keywords / control
37 QTextCharFormat kwFmt;
38 kwFmt.setForeground( QColor( "#c678dd" ) );
39 kwFmt.setFontWeight( QFont::Bold );
40 for ( const QString kw : QStringList{ "if", "else", "isnull", "null", "eval",
41 "graph", "string" } )
42 mRules.append( { QRegularExpression( "\\b" + kw + "\\b" ), kwFmt } );
43
44 // Functions
45 QTextCharFormat fnFmt;
46 fnFmt.setForeground( QColor( "#61afef" ) );
47 const QStringList funcs = {
48 "abs", "atan", "cos", "exp", "int", "log", "round",
49 "sin", "sqrt", "tan", "float", "double", "rand",
50 "min", "max", "median", "mode", "nmedian",
51 "ewres", "nsres", "row", "col", "x", "y",
52 "acos", "asin", "pow", "hypot", "fabs"
53 };
54 for ( const QString &f : funcs )
55 mRules.append( { QRegularExpression( "\\b" + f + "\\b(?=\\s*\\()" ), fnFmt } );
56
57 // Numeric literals
58 QTextCharFormat numFmt;
59 numFmt.setForeground( QColor( "#d19a66" ) );
60 mRules.append( { QRegularExpression( "\\b[0-9]+(\\.[0-9]+)?\\b" ), numFmt } );
61
62 // String literals (quoted map names or strings)
63 QTextCharFormat strFmt;
64 strFmt.setForeground( QColor( "#98c379" ) );
65 mRules.append( { QRegularExpression( "\"[^\"]*\"" ), strFmt } );
66
67 // Operators
68 QTextCharFormat opFmt;
69 opFmt.setForeground( QColor( "#e06c75" ) );
70 mRules.append( { QRegularExpression( "[+\\-*/^%=<>!&|]+" ), opFmt } );
71
72 // Output map name (left side of =)
73 QTextCharFormat outFmt;
74 outFmt.setForeground( QColor( "#e5c07b" ) );
75 outFmt.setFontWeight( QFont::Bold );
76 mRules.append( { QRegularExpression( "^\\s*[A-Za-z_][A-Za-z0-9_.@]*\\s*(?==)" ), outFmt } );
77
78 // Comments
79 QTextCharFormat cmt;
80 cmt.setForeground( QColor( "#5c6370" ) );
81 cmt.setFontItalic( true );
82 mRules.append( { QRegularExpression( "#[^\n]*" ), cmt } );
83 }
84
85 protected:
86 void highlightBlock( const QString &text ) override
87 {
88 for ( const auto &rule : mRules )
89 {
90 QRegularExpressionMatchIterator it = rule.pattern.globalMatch( text );
91 while ( it.hasNext() )
92 {
93 const QRegularExpressionMatch m = it.next();
94 setFormat( m.capturedStart(), m.capturedLength(), rule.format );
95 }
96 }
97 }
98
99 private:
100 struct Rule { QRegularExpression pattern; QTextCharFormat format; };
101 QList<Rule> mRules;
102};

258// Refresh available raster map list

260QStringList GrassMapCalcDialog::availableRasters() const
261{
262 // Use g.list to get rasters in all accessible mapsets
263 QProcess proc;
264 proc.setProcessEnvironment( GrassEnv::instance()->buildSubprocessEnv() );
265
266 const QString gisBase = QString::fromLocal8Bit( qgetenv( "GISBASE" ) );
267 if ( gisBase.isEmpty() ) return {};
268
269 proc.start( gisBase + "/bin/g.list",
270 { "type=raster", "mapset=.", "separator=newline" } );
271 proc.waitForFinished( 5000 );
272
273 return QString::fromLocal8Bit( proc.readAllStandardOutput() )
274 .split( '\n', Qt::SkipEmptyParts );
275}
