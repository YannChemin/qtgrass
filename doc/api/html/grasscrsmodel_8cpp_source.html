44void GrassCrsModel::loadAll()
45{
46 mAll.clear();
47
48 const QString authFilter = ( mAuth == EPSG ) ? "EPSG" : "IAU_2015";
49
50 // For IAU (planetary bodies) only geographic CRS are useful for location
51 // creation — the projected_crs table has 2000+ entries that swamp the list.
52 // For EPSG we include both geographic and projected (e.g. UTM zones).
53 // Wrapped in a subquery so ORDER BY CAST(code AS INTEGER) works in SQLite
54 // (UNION does not allow expressions in ORDER BY, only column names/positions).
55 const QString sql = ( mAuth == IAU )
56 ? R"(
57 SELECT auth_name, code, name, 'geographic' AS type
58 FROM geodetic_crs
59 WHERE auth_name = ?
60 AND (deprecated IS NULL OR deprecated = 0)
61 ORDER BY CAST(code AS INTEGER)
62 )"
63 : R"(
64 SELECT * FROM (
65 SELECT auth_name, code, name, 'geographic' AS type
66 FROM geodetic_crs
67 WHERE auth_name = ?
68 AND (deprecated IS NULL OR deprecated = 0)
69 UNION ALL
70 SELECT auth_name, code, name, 'projected' AS type
71 FROM projected_crs
72 WHERE auth_name = ?
73 AND (deprecated IS NULL OR deprecated = 0)
74 ) ORDER BY CAST(code AS INTEGER)
75 )";
76
77 QSqlQuery q( mDb );
78 q.prepare( sql );
79 q.addBindValue( authFilter );
80 if ( mAuth != IAU )
81 q.addBindValue( authFilter );
82
83 if ( !q.exec() )
84 {
85 qCWarning(lcCrsModel) << "GrassCrsModel query failed:" << q.lastError().text();
86 return;
87 }
88
89 while ( q.next() )
90 {
91 CrsEntry e;
92 e.authName = q.value( 0 ).toString();
93 e.code = q.value( 1 ).toString();
94 e.name = q.value( 2 ).toString();
95 e.type = q.value( 3 ).toString();
96 mAll.append( e );
97 }
98
99 // Build initial visible set (all)
100 mVisible.resize( mAll.size() );
101 std::iota( mVisible.begin(), mVisible.end(), 0 );
102}

105{
