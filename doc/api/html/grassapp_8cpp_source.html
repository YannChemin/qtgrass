551// ---------------------------------------------------------------------------
552// applyMenuScrollLimits — walk every QMenu reachable from the menu bar and
553// cap its maximum height. When a QMenu's sizeHint().height() exceeds
554// maximumHeight(), Qt automatically inserts its built-in up/down scroll
555// arrow items — no manual scrollbar management needed.
556// ---------------------------------------------------------------------------
557static void limitMenuHeight( QMenu *menu, int maxH )
558{
559 if ( !menu ) return;
560 menu->setMaximumHeight( maxH );
561 for ( QAction *a : menu->actions() )
562 if ( a->menu() )
563 limitMenuHeight( a->menu(), maxH );
564}

566void GrassApp::applyMenuScrollLimits()
567{
568 // Use 70 % of the primary screen's usable height.
569 // Minimum 400 px so tiny screens still get a usable menu.
570 const QRect screen = QGuiApplication::primaryScreen()->availableGeometry();
571 const int maxH = qMax( 400, static_cast<int>( screen.height() * 0.70 ) );
572
573 for ( QAction *a : menuBar()->actions() )
574 if ( a->menu() )
575 limitMenuHeight( a->menu(), maxH );
576}

715void GrassApp::loadWorkspace( const QString &path )
716{
717 GrassWorkspace ws;
718 if ( !ws.load( path ) )
719 {
720 mMessageBar->pushMessage( "Workspace",
721 "Cannot load " + path + ": " + ws.lastError(),
722 GrassMessageBar::Critical, 6 );
723 return;
724 }
725
726 // Clear tree + canvas together before switching session.
727 // mLayerPanel->clear() removes tree items and canvas layers atomically —
728 // no dangling pointers in the tree, and waitForFinished() ensures no render
729 // is running when openMapset() changes the GRASS global state.
730 mLayerPanel->clear();
731
732 // Switch session if the workspace encodes a different one
733 if ( !ws.database().isEmpty() && !ws.location().isEmpty() && !ws.mapset().isEmpty() )
734 {
735 if ( ws.database() != mEnv->gisDatabase() ||
736 ws.location() != mEnv->location() ||
737 ws.mapset() != mEnv->mapset() )
738 {
739 mEnv->openMapset( ws.database(), ws.location(), ws.mapset() );
740 // Reset viewport to the new session's region so the first render
741 // of the new session does not use the old session's coordinates
742 // (e.g. projected meters passed to a geographic location → crash).
743 mCanvas->refreshRegionCache();
744 mCanvas->resetViewportToRegion();
745 }
746 }
747
748 // Add layers from the first display (others are ignored for now)
749 if ( !ws.displays().isEmpty() )
750 {
751 const GwxDisplay &disp = ws.displays().first();
752
753 // Add in reverse order so the first layer ends up on top
754 for ( int i = disp.layers.size() - 1; i >= 0; --i )
755 {
756 const GwxLayer &l = disp.layers[i];
757 // Parse "map@mapset" → name, mapset
758 const int at = l.name.indexOf('@');
759 const QString mapName = (at >= 0) ? l.name.left(at) : l.name;
760 const QString mapMapset = (at >= 0) ? l.name.mid(at+1) : mEnv->mapset();
761
762 if ( l.type == "raster" || l.displayCmd == "d.rast" )
763 mLayerPanel->addRasterLayer( mapName, mapMapset );
764 else if ( l.type == "vector" || l.displayCmd == "d.vect" )
765 mLayerPanel->addVectorLayer( mapName, mapMapset );
766 }
767
768 // Restore the saved display viewport from the .gxw rather than
769 // zooming to layer extents. Layer auto-zoom (addLayer for size==1)
770 // runs first, so we override it with the saved extent afterwards.
771 //
772 // Priority:
773 // 1. Saved extent in .gxw (west,south,east,north from the display element)
774 // 2. GRASS computational region (WIND) — the session's working area
775 // 3. zoomToFullExtent() — last resort, dominated by largest raster
776 {
777 GrassExtent savedVp { disp.north, disp.south, disp.east, disp.west };
778 if ( savedVp.isValid() )
779 {
780 // Apply after event loop processes addLayer() signals (50 ms)
781 QTimer::singleShot( 50, mCanvas, [this, savedVp] {
782 mCanvas->zoomToExtent( savedVp );
783 } );
784 }
785 else
786 {
787 // No saved extent — use the GRASS computational region (WIND).
788 // resetViewportToRegion() is a no-op if mRegion is not yet valid,
789 // so the layer auto-zoom from addLayer() is kept as last resort.
790 QTimer::singleShot( 50, mCanvas, [this] {
791 mCanvas->resetViewportToRegion();
792 mCanvas->scheduleRender();
793 } );
794 }
795 }
796 }
797
798 // Explicitly refresh the data catalog after session switch.
799 // mapsetChanged already connects to refresh() but an explicit call ensures
800 // the catalog shows the new location even if the signal fired before the
801 // OS environment variable was fully visible to qgetenv().
802 if ( mCatalog ) mCatalog->refresh();
803
804 mMessageBar->pushMessage( "Workspace loaded",
805 QFileInfo( path ).fileName() +
806 QString( " — %1 layer(s)" ).arg(
807 ws.displays().isEmpty() ? 0 : ws.displays().first().layers.size() ),
808 GrassMessageBar::Info, 5 );
809}

811void GrassApp::saveWorkspace( const QString &path )
812{
813 // Write a minimal GRASS .gxw XML that wxPython and qtgrass can both open.
814 // Format:
815 // <gxw>
816 // <session> database / location / mapset </session>
817 // <display extent="W,S,E,N,0,1">
818 // <layer type="raster|vector" name="map@mapset" checked="1" opacity="1.0">
819 // <task name="d.rast|d.vect">
820 // <parameter name="map"><value>map@mapset</value></parameter>
821 // </task>
822 // </layer>
823 // </display>
824 // </gxw>
825 QFile f( path );
826 if ( !f.open( QIODevice::WriteOnly | QIODevice::Text ) )
827 {
828 mMessageBar->pushMessage( "Save Workspace",
829 "Cannot write to: " + path, GrassMessageBar::Critical, 6 );
830 return;
831 }
832
833 const GrassExtent vp = mCanvas->viewportExtent();
834 QTextStream out( &f );
835 out.setEncoding( QStringConverter::Utf8 );
836
837 out << "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n"
838 << "<!DOCTYPE gxw SYSTEM \"grass-gxw.dtd\">\n"
839 << "<gxw>\n"
840 << " <session>\n"
841 << " <database>" << mEnv->gisDatabase() << "</database>\n"
842 << " <location>" << mEnv->location() << "</location>\n"
843 << " <mapset>" << mEnv->mapset() << "</mapset>\n"
844 << " </session>\n"
845 << " <display name=\"Map Display 1\" render=\"1\" mode=\"0\" "
846 << "showCompExtent=\"1\" alignExtent=\"1\" constrainRes=\"0\" "
847 << "showStatusbar=\"1\" showToolbars=\"1\" isDocked=\"1\" "
848 << QString( "extent=\"%1,%2,%3,%4,0.000000,1.000000\" " )
849 .arg( vp.west, 0, 'f', 6 ).arg( vp.south, 0, 'f', 6 )
850 .arg( vp.east, 0, 'f', 6 ).arg( vp.north, 0, 'f', 6 )
851 << "viewMode=\"2d\">\n";
852
853 // Write layers in reverse order (bottom layer first in the XML, matching
854 // wxPython convention — qtgrass loadWorkspace() re-reverses on load)
855 const auto &layers = mCanvas->layers();
856 for ( int i = layers.size() - 1; i >= 0; --i )
857 {
858 const GrassMapLayer *l = layers[i];
859 const bool isRaster = dynamic_cast<const GrassRasterMapLayer *>( l ) != nullptr;
860 const QString type = isRaster ? "raster" : "vector";
861 const QString dcmd = isRaster ? "d.rast" : "d.vect";
862 // Build full map@mapset name
863 const QString fullName = l->name().contains('@')
864 ? l->name()
865 : l->name() + "@" + mEnv->mapset();
866 const int checked = l->visible() ? 1 : 0;
867 const double opacity = l->opacity();
868
869 out << QString( " <layer type=\"%1\" name=\"%2\" "
870 "checked=\"%3\" opacity=\"%4\">\n" )
871 .arg( type ).arg( fullName ).arg( checked ).arg( opacity, 0, 'f', 6 );
872 out << QString( " <task name=\"%1\">\n" ).arg( dcmd );
873 out << " <parameter name=\"map\">\n"
874 << " <value>" << fullName << "</value>\n"
875 << " </parameter>\n"
876 << " </task>\n"
877 << " </layer>\n";
878 }
879
880 out << " </display>\n"
881 << "</gxw>\n";
882
883 f.close();
884
885 QSettings().setValue( "session/lastWorkspace", path );
886 mMessageBar->pushMessage( "Workspace saved",
887 QFileInfo( path ).fileName() +
888 QString( " — %1 layer(s)" ).arg( layers.size() ),
889 GrassMessageBar::Info, 4 );
890}

912void GrassApp::applyTileSettings( bool enabled, const TileProvider &prov )
913{
914 // Remove existing tile layer from canvas if present
915 if ( mTileBackground )
916 {
917 mCanvas->setLayerVisible( mTileBackground->name(), false );
918 // Remove it from mCanvas's internal layer list.
919 // We use clearLayers() + re-add is too disruptive; instead we rely on
920 // the canvas ignoring invisible layers. For a clean swap, re-insert.
921 // Simplest: keep the layer object, just update provider and visibility.
922 }
923
924 if ( !enabled )
925 {
926 if ( mTileBackground )
927 {
928 mTileBackground->setVisible( false );
929 mTileBackground->invalidateCache();
930 }
931 mCanvas->scheduleRender();
932 return;
933 }
934
935 // Detect if location is geographic (lat/lon) — needed for tile math.
936 // G_projection() is safe to call on the main thread.
937 const bool isGeo = ( G_projection() == PROJECTION_LL );
938
939 mTileCache->setProviderId( prov.id );
940 mTileFetcher->setUrlTemplate( prov.urlTemplate );
941
942 if ( !mTileBackground )
943 {
944 // First time: create the layer and insert at bottom of stack.
945 mTileBackground = new GrassTileBackground(
946 mTileCache, mTileFetcher, isGeo, prov.attribution );
947 mTileBackground->setShowAttribution( true );
948
949 // Insert at index 0 (bottom-most) in the canvas.
950 mCanvas->insertLayerAt( 0, mTileBackground );
951
952 // When tiles arrive → invalidate tile layer cache + re-render.
953 connect( mTileFetcher, &GrassTileFetcher::tileFetched,
954 this, [this]( GrassTileKey ) {
955 if ( mTileBackground )
956 mTileBackground->invalidateCache();
957 mCanvas->scheduleRender();
958 } );
959 }
960 else
961 {
962 mTileBackground->setAttribution( prov.attribution );
963 mTileBackground->setVisible( true );
964 mTileBackground->invalidateCache();
965 }
966
967 if ( !isGeo )
968 {
969 mMessageBar->pushMessage( "Basemap",
970 "Projected CRS — tiles will be reprojected on-the-fly to the current location's coordinate system.",
971 GrassMessageBar::Info, 5 );
972 }
973
974 mCanvas->scheduleRender();
975}
