20// Scan GRASS databases for .gxw workspace files.
21// Returns a list of absolute paths.

23static QStringList findWorkspaceFiles( const QStringList &databases )
24{
25 QStringList found;
26 QStringList visited;
27
28 for ( const QString &db : databases )
29 {
30 const QString absDb = QDir( db ).absolutePath();
31 if ( absDb.isEmpty() || visited.contains( absDb ) ) continue;
32 visited << absDb;
33
34 const QDir dbDir( absDb );
35 if ( !dbDir.exists() ) continue;
36
37 // Each sub-directory of the db that has a PERMANENT/WIND is a location
38 for ( const QString &loc : dbDir.entryList( QDir::Dirs | QDir::NoDotAndDotDot ) )
39 {
40 const QDir locDir( absDb + "/" + loc );
41 if ( !QFileInfo( locDir.filePath( "PERMANENT/WIND" ) ).exists() )
42 continue; // not a GRASS location
43
44 for ( const QFileInfo &fi : locDir.entryInfoList( { "*.gxw" }, QDir::Files ) )
45 found << fi.absoluteFilePath();
46 }
47 }
48 found.sort( Qt::CaseInsensitive );
49 return found;
50}

161void GrassStartupDialog::openSelectedWorkspace()
162{
163 if ( !mWorkspaceTable ) return;
164 const int row = mWorkspaceTable->currentRow();
165 if ( row < 0 ) return;
166 const QTableWidgetItem *item = mWorkspaceTable->item( row, 0 );
167 if ( !item ) return;
168 const QString path = item->data( Qt::UserRole ).toString();
169 if ( path.isEmpty() ) return;
170 mSelectedWorkspace = path;
