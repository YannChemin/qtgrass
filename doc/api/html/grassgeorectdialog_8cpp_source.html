284
285 for (int r = 0; r < mGcps.size(); ++r)

289 auto *item = new QTableWidgetItem(text);

339void GrassGeoRectDialog::updateResiduals()
340{
341 const int order = mOrderCombo->currentData().toInt();
342 const bool ok = mTransform.fit(mGcps, order);
343
344 if (ok)
345 {
346 mTransform.computeResiduals(mGcps);
347 const double totalRms = mTransform.totalRms(mGcps);
348 mRmsLabel->setText(QString("Total RMS: %1 (order %2, %3 active GCPs)")
349 .arg(totalRms, 0, 'f', 4)
350 .arg(order)
351 .arg([&]{ int n=0; for(auto &g:mGcps) if(g.enabled) ++n; return n; }()));
352 }
353 else
354 {
355 for (auto &g : mGcps) g.resX = g.resY = g.rms = 0;
356 mRmsLabel->setText(
357 QString("RMS: — (need ≥ %1 active GCPs for order %2)")
358 .arg(mTransform.minPoints()).arg(order));
359 }
360
361 // Refresh residual columns without full rebuild
362 mUpdatingTable = true;
363 for (int r = 0; r < mGcps.size(); ++r)
364 {
365 const auto &g = mGcps[r];
366 if (auto *it = mGcpTable->item(r, COL_RESX))
367 it->setText(QString::number(g.resX, 'f', 4));
368 if (auto *it = mGcpTable->item(r, COL_RESY))
369 it->setText(QString::number(g.resY, 'f', 4));
370 if (auto *it = mGcpTable->item(r, COL_RMS))
371 {
372 it->setText(g.enabled ? QString::number(g.rms, 'f', 4) : "—");
373 if (g.enabled && g.rms > 0)
374 {
375 const int red = qMin(255, static_cast<int>(g.rms * 100));
376 const int green = qMax(0, 255 - red);
377 it->setBackground(QColor(red, green, 0, 80));
378 }
379 }
380 }
381 mUpdatingTable = false;

394 // Query GRASS environment for the group POINTS file path
395 QProcess p;

397 if (!p.waitForFinished(5000)) return {};

400 if (parts.size() < 3) return {};

403 parts[2].trimmed(), mGroupEdit->text().trimmed());

406void GrassGeoRectDialog::savePoints()
407{
408 QString path = pointsFilePath();
409 if (path.isEmpty())
410 path = QFileDialog::getSaveFileName(this, "Save POINTS file", {},
411 "GRASS POINTS (*)");
412 if (path.isEmpty()) return;
413
414 // Ensure parent directory exists
415 QDir().mkpath(QFileInfo(path).absolutePath());
416
417 QFile f(path);
418 if (!f.open(QIODevice::WriteOnly | QIODevice::Text))
419 {
420 QMessageBox::warning(this, "Save failed", "Cannot write: " + path);
421 return;
422 }
423 QTextStream out(&f);
424 out << "# Ground Control Points File\n";
425 out << "# ref_east ref_north src_east src_north status(1=active)\n";
426 for (const GcpPoint &g : mGcps)
427 out << QString("%1 %2 %3 %4 %5\n")
428 .arg(g.refX, 0, 'f', 8).arg(g.refY, 0, 'f', 8)
429 .arg(g.srcX, 0, 'f', 8).arg(g.srcY, 0, 'f', 8)
430 .arg(g.enabled ? 1 : 0);
431
432 log("Saved " + QString::number(mGcps.size()) + " GCPs → " + path);
433}

435void GrassGeoRectDialog::loadPoints()
436{
437 QString path = pointsFilePath();
438 if (!QFile::exists(path))
439 path = QFileDialog::getOpenFileName(this, "Load POINTS file", {},
440 "GRASS POINTS (*);;All Files (*)");
441 if (path.isEmpty() || !QFile::exists(path)) return;
442
443 QFile f(path);
444 if (!f.open(QIODevice::ReadOnly | QIODevice::Text)) return;
445
446 mGcps.clear();
447 mNextId = 1;
448 QTextStream in(&f);
449 while (!in.atEnd())
450 {
451 const QString line = in.readLine().trimmed();
452 if (line.isEmpty() || line.startsWith('#')) continue;
453 const QStringList cols = line.split(' ', Qt::SkipEmptyParts);
454 if (cols.size() < 4) continue;
455 GcpPoint g;
456 g.id = mNextId++;
457 g.refX = cols[0].toDouble();
458 g.refY = cols[1].toDouble();
459 g.srcX = cols[2].toDouble();
460 g.srcY = cols[3].toDouble();
461 g.enabled = (cols.size() < 5 || cols[4].toInt() != 0);
462 mGcps.append(g);
463 }
464 updateGcpTable();
465 updateResiduals();
466 refreshMarkers();
467 log("Loaded " + QString::number(mGcps.size()) + " GCPs from " + path);
468}

491void GrassGeoRectDialog::rectify()
492{
493 if (mGcps.size() < 3)
494 {
495 QMessageBox::warning(this, "Rectify",
496 "Add at least 3 GCPs before rectifying.");
497 return;
498 }
499
500 const int order = mOrderCombo->currentData().toInt();
501 if (!mTransform.fit(mGcps, order))
502 {
503 QMessageBox::warning(this, "Rectify",
504 QString("Need ≥ %1 active GCPs for order %2 polynomial.")
505 .arg(mTransform.minPoints()).arg(order));
506 return;
507 }
508
509 // Save the POINTS file first
510 savePoints();
511
512 // Create group
513 createGroup();
514
515 // Run i.rectify
516 const QString grp = mGroupEdit->text().trimmed();
517 const QString ext = mExtensionEdit->text().trimmed();
518 const QString method = mMethodCombo->currentText();
519
520 QStringList args = {"group=" + grp,
521 "extension=" + ext,
522 "order=" + QString::number(order),
523 "method=" + method,
524 "-c"}; // -c = use the group's POINTS file
525
526 log("Running: i.rectify " + args.join(' '));
527
528 QProcess p;
529 p.setProcessChannelMode(QProcess::MergedChannels);
530 p.start("i.rectify", args);
531
532 if (!p.waitForFinished(300000))
533 {
534 log("i.rectify timed out.", true);
535 return;
536 }
537
538 const QString out = QString::fromLocal8Bit(p.readAll()).trimmed();
539 if (p.exitCode() != 0)
540 {
541 log("i.rectify failed:\n" + out, true);
542 QMessageBox::critical(this, "Rectification failed", out.left(400));
543 return;
544 }
545
546 log("Rectification complete. Output maps have extension '" + ext + "'.");
547 QMessageBox::information(this, "Rectification complete",
548 QString("Maps rectified with extension '%1'.\n"
549 "Total RMS: %2\n\nOutput maps added to '%3'.")
550 .arg(ext).arg(mTransform.totalRms(mGcps), 0, 'f', 4).arg(grp));
551}
