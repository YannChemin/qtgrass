209void GrassMapsetBrowser::populateMapsets( const QString &location )
210{
211 // Scan the location directory for .gxw workspace files
212 if ( !mGisdbase.isEmpty() && !location.isEmpty() )
213 updateWorkspaceList( mGisdbase + "/" + location );
214 mMslist->clear();
215 mMapset.clear();
216 validate();
217
218 if ( mGisdbase.isEmpty() || location.isEmpty() ) return;
219
220 QDir locDir( mGisdbase + "/" + location );
221 for ( const QString &entry : locDir.entryList( QDir::Dirs | QDir::NoDotAndDotDot,
222 QDir::Name ) )
223 {
224 // A valid mapset contains a WIND file
225 if ( QFile::exists( locDir.filePath( entry + "/WIND" ) ) )
226 mMslist->addItem( entry );
227 }
228
229 // Auto-select PERMANENT if present
230 const auto perm = mMslist->findItems( "PERMANENT", Qt::MatchExactly );
231 if ( !perm.isEmpty() )
232 mMslist->setCurrentItem( perm.first() );
233}
