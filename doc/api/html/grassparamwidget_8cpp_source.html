46// Factory
47GrassParamWidget *GrassParamWidget::create( const GrassParamDesc &desc,
48 QWidget *parent )
49{
50 // Enum (discrete values list)
51 if ( !desc.values.isEmpty() )
52 return new GrassEnumParam( desc, parent );
53
54 // File prompt
55 if ( desc.hasGisprompt )
56 {
57 const QString &pr = desc.gisprompt.prompt;
58 if ( pr == "file" || pr == "datasource" || pr == "datasource_layer" )
59 return new GrassFileParam( desc, parent );
60 // raster / vector / layer / other map names → plain line edit
61 return new GrassLineEditParam( desc, parent );
62 }
63
64 // Numeric types
65 if ( desc.type == "integer" )
66 return new GrassSpinParam( desc, parent );
67 if ( desc.type == "float" || desc.type == "double" )
68 return new GrassDoubleSpinParam( desc, parent );
69
70 return new GrassLineEditParam( desc, parent );
71}

265 this, &GrassEnumParam::valueChanged );
