151bool GrassFieldCalcDialog::createColumn(const QString &name, const QString &type)
152{
153 QProcess p;
154 p.setProcessChannelMode(QProcess::MergedChannels);
155 p.start("v.db.addcolumn",
156 {"map=" + mConn.mapName,
157 "layer=" + QString::number(mConn.layer),
158 "columns=" + name + " " + type});
159 if (!p.waitForFinished(15000) || p.exitCode() != 0)
160 {
161 QMessageBox::warning(this, "Add column failed",
162 QString::fromLocal8Bit(p.readAll()).trimmed());
163 return false;
164 }

168void GrassFieldCalcDialog::execute()
169{
170 const QString expr = mExprEdit->toPlainText().trimmed();
171 if (expr.isEmpty())
172 {
173 QMessageBox::warning(this, "Field Calculator", "Expression is empty.");
174 return;
175 }
176
177 // Determine target column
178 QString targetCol;
179 if (mTargetCombo->currentData().toString() == "new")
180 {
181 targetCol = mNewNameEdit->text().trimmed();
182 if (targetCol.isEmpty())
183 {
184 QMessageBox::warning(this, "Field Calculator",
185 "Enter a name for the new column.");
186 return;
187 }
188 if (!createColumn(targetCol, mNewTypeCombo->currentText()))
189 return;
190 // Add the new column to the list
191 mColumns.append(targetCol);
192 }
193 else
194 {
195 targetCol = mTargetCombo->currentData().toString();
196 }
197
198 // Build v.db.update args
199 QStringList args = {"map=" + mConn.mapName,
200 "layer=" + QString::number(mConn.layer),
201 "column=" + targetCol,
202 "qcolumn=" + expr};
203
204 const QString where = mWhereEdit->text().trimmed();
205 if (!where.isEmpty())
206 args << "where=" + where;
207
208 QProcess p;
209 p.setProcessChannelMode(QProcess::MergedChannels);
210 p.start("v.db.update", args);
211
212 mRunBtn->setEnabled(false);
213 mRunBtn->setText("Running…");
214
215 const bool finished = p.waitForFinished(120000);
216 mRunBtn->setEnabled(true);
217 mRunBtn->setText("▶ Run");
218
219 if (!finished || p.exitCode() != 0)
220 {
221 const QString err = QString::fromLocal8Bit(p.readAll()).trimmed();
222 QMessageBox::warning(this, "Field Calculator failed",
223 err.isEmpty() ? "v.db.update returned non-zero exit code." : err);
224 return;
225 }
226
227 QMessageBox::information(this, "Field Calculator",
228 QString("Column '%1' updated successfully.").arg(targetCol));
229 emit calculationDone();
230}
