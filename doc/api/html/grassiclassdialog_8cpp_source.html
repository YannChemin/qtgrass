177 // List bands in the group
178 QProcess p;
179 p.setProcessChannelMode(QProcess::MergedChannels);

182 { log("i.group -l timed out", true); return; }

184 mBands.clear();

190 mBands.append(l);

197 }
198
199 mDisplayBand->clear();

203 for (const QString &b : mBands)

206 mDisplayBand->addItem(shortName, b);
207 mScatterBandX->addItem(shortName);
208 mScatterBandY->addItem(shortName);
209 labels << shortName;

337BandStats GrassIclassDialog::statsForBand(const IClassDef::BBox &bbox,
338 const QString &band)
339{
340 BandStats bs;
341
342 // Set region to bbox, aligned to band
343 {
344 QProcess p;
345 p.setProcessChannelMode(QProcess::MergedChannels);
346 p.start("g.region",
347 {QString("n=%1").arg(bbox.n, 0, 'f', 10),
348 QString("s=%1").arg(bbox.s, 0, 'f', 10),
349 QString("e=%1").arg(bbox.e, 0, 'f', 10),
350 QString("w=%1").arg(bbox.w, 0, 'f', 10),
351 "align=" + band});
352 p.waitForFinished(10000);
353 }
354
355 // Get stats using r.univar -g
356 QProcess p;
357 p.setProcessChannelMode(QProcess::MergedChannels);
358 p.start("r.univar", {"-g", "map=" + band});
359 if (!p.waitForFinished(30000)) return bs;
360
361 for (const QString &line :
362 QString::fromLocal8Bit(p.readAll()).split('\n', Qt::SkipEmptyParts))
363 {
364 if (line.startsWith("n=")) bs.count = line.mid(2).toInt();
365 if (line.startsWith("mean=")) bs.mean = line.mid(5).toDouble();
366 if (line.startsWith("population_stddev="))
367 bs.stdDev = line.mid(18).toDouble();
368 if (line.startsWith("stddev=") && bs.stdDev == 0)
369 bs.stdDev = line.mid(7).toDouble();
370 }
371 return bs;
372}

374void GrassIclassDialog::computeStats()
375{
376 if (mBands.isEmpty())
377 { log("Load an imagery group first.", true); return; }
378
379 // Save current region
380 QProcess().start("g.region", {"save=tmp_iclass_backup", "--overwrite"});
381
382 int totalAreas = 0;
383 for (auto &cls : mClasses)
384 {
385 cls.stats.resize(mBands.size());
386 for (int b = 0; b < mBands.size(); ++b)
387 {
388 BandStats combined;
389 for (const auto &area : cls.areas)
390 {
391 const BandStats s = statsForBand(area, mBands[b]);
392 // Accumulate: simple average of means (approximate)
393 if (s.count > 0)
394 {
395 combined.count += s.count;
396 combined.mean += s.mean;
397 combined.stdDev += s.stdDev;
398 }
399 }
400 if (!cls.areas.isEmpty() && combined.count > 0)
401 {
402 combined.mean /= cls.areas.size();
403 combined.stdDev /= cls.areas.size();
404 }
405 cls.stats[b] = combined;
406 }
407 totalAreas += cls.areas.size();
408
409 // Update pixel count in table
410 if (mSelectedClass >= 0 && mSelectedClass < mClasses.size())
411 {
412 const int row = &cls - mClasses.data();
413 if (auto *item = mClassTable->item(row, 3))
414 item->setText(cls.stats.value(0).count > 0
415 ? QString::number(cls.stats[0].count)
416 : "—");
417 }
418 }
419
420 // Restore region
421 {
422 QProcess p;
423 p.start("g.region", {"region=tmp_iclass_backup"});
424 p.waitForFinished(5000);
425 }
426 QProcess().start("g.remove", {"type=region", "name=tmp_iclass_backup", "-f"});
427
428 mScatter->setClasses(mClasses);
429 mScatter->setBandIndices(mScatterBandX->currentIndex(),
430 mScatterBandY->currentIndex());
431
432 // Update table
433 for (int r = 0; r < mClasses.size(); ++r)
434 {
435 const auto &cls = mClasses[r];
436 if (auto *item = mClassTable->item(r, 3))
437 item->setText(cls.stats.value(0).count > 0
438 ? QString::number(cls.stats[0].count) : "—");
439 }
440
441 log(QString("Statistics computed for %1 classes (%2 areas).")
442 .arg(mClasses.size()).arg(totalAreas));
443}

447bool GrassIclassDialog::buildTrainingMap(const QString &mapName)
448{
449 // Create per-class bbox rasters and merge them
450 QStringList classMaps;
451 for (const auto &cls : mClasses)
452 {
453 if (cls.areas.isEmpty()) continue;
454 for (int ai = 0; ai < cls.areas.size(); ++ai)
455 {
456 const auto &area = cls.areas[ai];
457 const QString tmpMap = QString("tmp_iclass_c%1_a%2").arg(cls.id).arg(ai);
458
459 // Set region to area, aligned to first band
460 QProcess p1;
461 p1.start("g.region",
462 {QString("n=%1").arg(area.n, 0, 'f', 10),
463 QString("s=%1").arg(area.s, 0, 'f', 10),
464 QString("e=%1").arg(area.e, 0, 'f', 10),
465 QString("w=%1").arg(area.w, 0, 'f', 10),
466 "align=" + mBands[0]});
467 p1.waitForFinished(10000);
468
469 // Create constant raster = class id
470 QProcess p2;
471 p2.setProcessChannelMode(QProcess::MergedChannels);
472 p2.start("r.mapcalc",
473 {QString("expression=%1 = %2").arg(tmpMap).arg(cls.id),
474 "--overwrite"});
475 if (!p2.waitForFinished(30000) || p2.exitCode() != 0)
476 {
477 log("r.mapcalc failed: " + QString::fromLocal8Bit(p2.readAll()), true);
478 return false;
479 }
480 classMaps.append(tmpMap);
481 }
482 }
483
484 if (classMaps.isEmpty())
485 {
486 log("No training areas defined.", true);
487 return false;
488 }
489
490 // Restore original region (default) for merging
491 QProcess().start("g.region", {"-d"}); // use default region
492 QThread::msleep(200);
493
494 // Build merge expression: patch all class maps
495 QString expr = classMaps[0];
496 for (int i = 1; i < classMaps.size(); ++i)
497 expr = QString("if(!isnull(%1), %1, %2)").arg(classMaps[i - 1], expr);
498 // Actually patch() would be cleaner, but r.mapcalc's if/isnull works
499 // Correct chained merge: use r.patch
500 QProcess pPatch;
501 pPatch.setProcessChannelMode(QProcess::MergedChannels);
502 pPatch.start("r.patch",
503 {"input=" + classMaps.join(","),
504 "output=" + mapName,
505 "--overwrite"});
506 if (!pPatch.waitForFinished(60000) || pPatch.exitCode() != 0)
507 {
508 log("r.patch failed: " + QString::fromLocal8Bit(pPatch.readAll()), true);
509 return false;
510 }
511
512 // Clean up temp maps
513 QProcess().start("g.remove",
514 QStringList{"type=raster", "name=" + classMaps.join(","), "-f"});
515 return true;
516}

518void GrassIclassDialog::runClassify()
519{
520 if (mBands.isEmpty() || mClasses.isEmpty())
521 { log("Load a group and define classes first.", true); return; }
522
523 const QString grp = mGroupCombo->currentText().trimmed();
524 const QString method = mMethodCombo->currentText();
525 const QString output = mOutputEdit->text().trimmed();
526
527 if (output.isEmpty())
528 { log("Set an output map name.", true); return; }
529
530 // Save + compute stats if not done
531 computeStats();
532
533 // Save current region, build training map
534 QProcess().start("g.region", {"save=tmp_iclass_backup", "--overwrite"});
535 QThread::msleep(300);
536
537 const QString trainMap = "tmp_iclass_trainingmap";
538 if (!buildTrainingMap(trainMap))
539 {
540 QProcess().start("g.region", {"region=tmp_iclass_backup"});
541 return;
542 }
543
544 // Restore region
545 QProcess p0; p0.start("g.region", {"region=tmp_iclass_backup"});
546 p0.waitForFinished(5000);
547 QProcess().start("g.remove", {"type=region", "name=tmp_iclass_backup", "-f"});
548
549 // Generate signature file
550 const QString sigFile = "iclass_sig_" + output;
551 log("Generating signatures with i.gensig…");
552 {
553 QProcess p;
554 p.setProcessChannelMode(QProcess::MergedChannels);
555 p.start("i.gensig",
556 {"trainingmap=" + trainMap,
557 "group=" + grp,
558 "subgroup=" + grp,
559 "signaturefile=" + sigFile});
560 if (!p.waitForFinished(120000) || p.exitCode() != 0)
561 {
562 log("i.gensig failed: " + QString::fromLocal8Bit(p.readAll()).trimmed(), true);
563 return;
564 }
565 }
566
567 // Run classifier
568 const QString modName = (method == "smap") ? "i.smap" : "i.maxlik";
569 log("Running " + modName + "…");
570 {
571 QProcess p;
572 p.setProcessChannelMode(QProcess::MergedChannels);
573 QStringList args = {"group=" + grp,
574 "subgroup=" + grp,
575 "signaturefile=" + sigFile,
576 "output=" + output,
577 "--overwrite"};
578 p.start(modName, args);
579 if (!p.waitForFinished(600000))
580 { log(modName + " timed out (>10 min)", true); return; }
581 if (p.exitCode() != 0)
582 { log(modName + " failed: " + QString::fromLocal8Bit(p.readAll()).trimmed(), true); return; }
583 }
584
585 // Clean up training map
586 QProcess().start("g.remove", {"type=raster", "name=" + trainMap, "-f"});
587
588 log("Classification complete → " + output);
589 QMessageBox::information(this, "Classification complete",
590 QString("Output map: %1\nMethod: %2\nClasses: %3")
591 .arg(output, method).arg(mClasses.size()));
592}
