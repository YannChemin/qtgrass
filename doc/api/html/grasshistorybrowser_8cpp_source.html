90void GrassHistoryBrowserDialog::loadHistory()
91{
92 const QString type = mTypeCombo->currentText();
93 const QString map = mMapCombo->currentText().trimmed();
94 if (map.isEmpty()) return;
95
96 const QString prog = (type == "vector") ? "v.info"
97 : (type == "raster_3d") ? "r3.info"
98 : "r.info";
99
100 QProcess p;
101 p.setProcessChannelMode(QProcess::MergedChannels);
102 p.start(prog, {"-h", "map=" + map});
103 if (!p.waitForFinished(15000))
104 { mStatusLabel->setText("Timed out."); return; }
105
106 const QString out = QString::fromLocal8Bit(p.readAll()).trimmed();
107 if (p.exitCode() != 0 || out.isEmpty())
108 {
109 mStatusLabel->setText("No history for map: " + map);
110 mHistoryEdit->clear();
111 return;
112 }
113
114 mHistoryEdit->setPlainText(out);
115 mStatusLabel->setText(QString("History for %1 '%2'").arg(type, map));
116}

118void GrassHistoryBrowserDialog::reRunSelected()
119{
120 const QString text = mHistoryEdit->toPlainText().trimmed();
121 if (text.isEmpty()) { mStatusLabel->setText("Load history first."); return; }
122
123 // Find the first non-comment line that looks like a GRASS command
124 // History lines typically start with a module name like "r.slope.aspect"
125 static const QRegularExpression moduleRe(
126 R"(^\s*(r\.|v\.|g\.|d\.|i\.|t\.|db\.|r3\.|m\.|ps\.|s\.\w+))",
127 QRegularExpression::MultilineOption);
128
129 QString cmd;
130 const auto match = moduleRe.match(text);
131 if (match.hasMatch())
132 {
133 // Extract the line containing the match
134 const int lineStart = text.lastIndexOf('\n', match.capturedStart()) + 1;
135 int lineEnd = text.indexOf('\n', match.capturedStart());
136 if (lineEnd < 0) lineEnd = text.size();
137 cmd = text.mid(lineStart, lineEnd - lineStart).trimmed();
138 }
139
140 if (cmd.isEmpty())
141 {
142 mStatusLabel->setText("No recognisable GRASS command found in history.");
143 return;
144 }
145
146 // Parse: first token = module name, rest = args
147 const QStringList parts = cmd.split(QRegularExpression("\\s+"), Qt::SkipEmptyParts);
148 if (parts.isEmpty()) return;
149
150 const QString module = parts[0];
151 const QStringList args = parts.mid(1);
152
153 mStatusLabel->setText("Opening: " + cmd.left(80));
154 auto *dlg = new GrassModuleDialog(module, args, mIface, this);
155 dlg->setAttribute(Qt::WA_DeleteOnClose);
