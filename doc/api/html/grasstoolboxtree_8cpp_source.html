28bool GrassToolboxFilterProxy::filterAcceptsRow( int sourceRow,
29 const QModelIndex &sourceParent ) const
30{
31 // With setRecursiveFilteringEnabled(true), Qt handles child propagation.
32 // We only need to match on the Search role.
33 const QModelIndex idx = sourceModel()->index( sourceRow, 0, sourceParent );
34 const QString searchText = idx.data( Qt::UserRole + 1 ).toString();
35 const QRegularExpression re = filterRegularExpression();
36 return re.pattern().isEmpty() || re.match( searchText ).hasMatch();
37}

39bool GrassToolboxFilterProxy::anyChildMatches( const QModelIndex &parent ) const
40{
41 // Not called when setRecursiveFilteringEnabled is on, kept for clarity
42 const int rows = sourceModel()->rowCount( parent );
43 for ( int r = 0; r < rows; ++r )
44 {
45 const QModelIndex child = sourceModel()->index( r, 0, parent );
46 if ( filterAcceptsRow( r, parent ) ) return true;
47 if ( sourceModel()->hasChildren( child ) && anyChildMatches( child ) ) return true;
48 }
49 return false;
50}

178void GrassToolboxTree::populateModel( const QDomElement &items,
179 QStandardItem *parent )
180{
181 QDomElement child = items.firstChildElement();
182 while ( !child.isNull() )
183 {
184 const QString tag = child.tagName();
185
186 if ( tag == "menu" )
187 {
188 // Sub-group
189 const QString label = child.firstChildElement( "label" ).text().trimmed()
190 .remove( '&' );
191
192 QStandardItem *groupItem = new QStandardItem( label );
193 groupItem->setData( "", static_cast<int>( Role::Command ) );
194 groupItem->setData( label.toLower(), static_cast<int>( Role::Search ) );
195 groupItem->setEditable( false );
196
197 QFont f = groupItem->font();
198 f.setItalic( true );
199 groupItem->setFont( f );
200
201 const QDomElement subItems = child.firstChildElement( "items" );
202 if ( !subItems.isNull() )
203 populateModel( subItems, groupItem );
204
205 parent->appendRow( groupItem );
206 }
207 else if ( tag == "menuitem" )
208 {
209 const QString label = child.firstChildElement( "label" ).text().trimmed();
210 const QString command = child.firstChildElement( "command" ).text().trimmed();
211 const QString help = child.firstChildElement( "help" ).text().trimmed();
212 const QString kw = child.firstChildElement( "keywords" ).text().trimmed();
213
214 if ( command.isEmpty() )
215 {
216 child = child.nextSiblingElement();
217 continue;
218 }
219
220 // Display: "label (command)" so both are visible without tooltip
221 const QString display = label.isEmpty()
222 ? command
223 : QString( "%1 (%2)" ).arg( label, command );
224
225 QStandardItem *item = new QStandardItem( display );
226 item->setData( command, static_cast<int>( Role::Command ) );
227
228 // Search role: label + command + keywords, lower-case, space-joined
229 const QString searchStr = ( label + " " + command + " " + kw ).toLower();
230 item->setData( searchStr, static_cast<int>( Role::Search ) );
231
232 item->setToolTip( help.isEmpty() ? command : help );
233 item->setEditable( false );
234
235 parent->appendRow( item );
236 }
237 // <separator/> and other tags are skipped
238
239 child = child.nextSiblingElement();
240 }
241}
