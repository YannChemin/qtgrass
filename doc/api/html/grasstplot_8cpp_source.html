35void TplotChart::paintEvent(QPaintEvent *)
36{
37 QPainter p(this);
38 p.setRenderHint(QPainter::Antialiasing);
39 p.fillRect(rect(), Qt::white);
40
41 const QRectF plot(55, 15, width() - 70, height() - 55);
42 p.fillRect(plot, QColor(248, 248, 252));
43 p.setPen(QPen(QColor(200, 200, 210)));
44 p.drawRect(plot);
45
46 if (mData.isEmpty())
47 {
48 p.setPen(Qt::gray);
49 QFont f; f.setPointSize(10); p.setFont(f);
50 p.drawText(plot, Qt::AlignCenter, "No data.\nSet coordinates and run the plot.");
51 return;
52 }
53
54 // Value range
55 double lo = std::numeric_limits<double>::max();
56 double hi = -std::numeric_limits<double>::max();
57 for (const auto &pt : mData)
58 {
59 if (!std::isnan(pt.second)) { lo = qMin(lo, pt.second); hi = qMax(hi, pt.second); }
60 }
61 if (lo >= hi) { lo -= 1; hi += 1; }
62 const double margin = (hi - lo) * 0.05;
63 lo -= margin; hi += margin;
64
65 const int n = mData.size();
66 auto mapX = [&](int i) { return plot.left() + (n > 1 ? i * plot.width() / (n - 1) : plot.width() / 2); };
67 auto mapY = [&](double v) { return plot.bottom() - (v - lo) / (hi - lo) * plot.height(); };
68
69 // Grid lines (5 horizontal)
70 QFont sf; sf.setPointSize(7); p.setFont(sf);
71 p.setPen(QPen(QColor(220, 220, 230), 0.5));
72 for (int i = 0; i <= 5; ++i)
73 {
74 const double v = lo + i * (hi - lo) / 5;
75 const double py = mapY(v);
76 p.drawLine(QPointF(plot.left(), py), QPointF(plot.right(), py));
77 p.setPen(Qt::black);
78 p.drawText(QRectF(2, py - 7, 50, 14), Qt::AlignRight | Qt::AlignVCenter,
79 QString::number(v, 'f', 2));
80 p.setPen(QPen(QColor(220, 220, 230), 0.5));
81 }
82
83 // Line + points
84 QPainterPath path;
85 bool first = true;
86 for (int i = 0; i < n; ++i)
87 {
88 if (std::isnan(mData[i].second)) { first = true; continue; }
89 const QPointF pt(mapX(i), mapY(mData[i].second));
90 if (first) { path.moveTo(pt); first = false; }
91 else path.lineTo(pt);
92 }
93 p.setPen(QPen(QColor(30, 100, 200), 2));
94 p.setBrush(Qt::NoBrush);
95 p.drawPath(path);
96
97 // Dots
98 p.setBrush(QColor(30, 100, 200));
99 p.setPen(QPen(Qt::white, 1));
100 for (int i = 0; i < n; ++i)
101 {
102 if (std::isnan(mData[i].second)) continue;
103 p.drawEllipse(QPointF(mapX(i), mapY(mData[i].second)), 4, 4);
104 }
105
106 // X-axis labels (sample every N)
107 p.setPen(Qt::black);
108 p.setFont(sf);
109 const int step = qMax(1, n / 8);
110 for (int i = 0; i < n; i += step)
111 {
112 const QString lbl = (i < mXLabels.size()) ? mXLabels[i].left(10)
113 : QString::number(i + 1);
114 p.save();
115 p.translate(mapX(i), plot.bottom() + 3);
116 p.rotate(30);
117 p.drawText(QRectF(0, 0, 70, 14), Qt::AlignLeft, lbl);
118 p.restore();
119 }
120
121 // Y-axis label
122 p.save();
123 p.translate(10, plot.center().y());
124 p.rotate(-90);
125 QFont yf; yf.setPointSize(8); p.setFont(yf);
126 p.drawText(QRectF(-plot.height()/2, -10, plot.height(), 20),
127 Qt::AlignCenter, "Value");
128 p.restore();
129}

192void GrassTplotDialog::runTplot()
193{
194 const QString strds = mStrdCombo->currentText().trimmed();
195 const QString east = mEastEdit->text().trimmed();
196 const QString north = mNorthEdit->text().trimmed();
197
198 if (strds.isEmpty() || east.isEmpty() || north.isEmpty())
199 {
200 mStatusLabel->setText("Fill in all fields.");
201 return;
202 }
203
204 // Step 1: get map list + timestamps
205 QProcess p1;
206 p1.setProcessChannelMode(QProcess::MergedChannels);
207 p1.start("t.rast.list",
208 {"input=" + strds, "separator=pipe",
209 "columns=name,start_time", "-u"});
210 if (!p1.waitForFinished(30000))
211 { mStatusLabel->setText("t.rast.list timed out."); return; }
212
213 struct Frame { QString name; QString date; };
214 QVector<Frame> frames;
215 for (const QString &line :
216 QString::fromLocal8Bit(p1.readAll()).split('\n', Qt::SkipEmptyParts))
217 {
218 const QStringList parts = line.split('|');
219 if (parts.size() < 1 || !parts[0].contains('@')) continue;
220 Frame f;
221 f.name = parts[0].trimmed();
222 f.date = (parts.size() > 1) ? parts[1].trimmed() : f.name;
223 frames.append(f);
224 }
225
226 if (frames.isEmpty())
227 { mStatusLabel->setText("No frames found in " + strds); return; }
228
229 mStatusLabel->setText(
230 QString("Querying %1 frames at E=%2, N=%3…").arg(frames.size()).arg(east, north));
231 QApplication::processEvents();
232
233 // Step 2: r.what for each frame
234 QVector<QPair<double,double>> data;
235 QStringList labels;
236 const QString coords = east + "," + north;
237 int done = 0;
238 for (const Frame &fr : frames)
239 {
240 QProcess p2;
241 p2.setProcessChannelMode(QProcess::MergedChannels);
242 p2.start("r.what",
243 {"map=" + fr.name,
244 "coordinates=" + coords,
245 "separator=pipe"});
246 p2.waitForFinished(10000);
247
248 // Output: east|north|site|value
249 const QString out = QString::fromLocal8Bit(p2.readAll()).trimmed();
250 double val = std::numeric_limits<double>::quiet_NaN();
251 const QStringList cols = out.split('|');
252 if (cols.size() >= 4)
253 {
254 bool ok;
255 const double v = cols[3].trimmed().toDouble(&ok);
256 if (ok) val = v;
257 }
258 data.append({ static_cast<double>(done++), val });
259 labels.append(fr.date.left(10));
260 }
261
262 mChart->setData(data, labels);
263 const long valid = std::count_if(data.begin(), data.end(),
264 [](const QPair<double,double> &pt){ return !std::isnan(pt.second); });
265 mStatusLabel->setText(
266 QString("%1 frames plotted (%2 valid values) for %3")
267 .arg(frames.size()).arg(valid).arg(strds));
268}
