31bool GrassModel::loadFile(const QString &path)
32{
33 QFile f(path);
34 if (!f.open(QIODevice::ReadOnly)) return false;
35
36 QDomDocument doc;
37 auto result = doc.setContent(&f);
38 f.close();
39 if (!result) return false;
40
41 QDomElement root = doc.documentElement();
42 if (root.tagName() != "model") return false;
43
44 clear();
45 mName = root.attribute("name", "Unnamed model");
46 mDescription = root.attribute("description", {});
47
48 auto intAttr = [](const QDomElement &e, const QString &n, int def = 0) {
49 return e.hasAttribute(n) ? e.attribute(n).toInt() : def;
50 };
51
52 for (QDomElement e = root.firstChildElement("action");
53 !e.isNull(); e = e.nextSiblingElement("action"))
54 {
55 ModelActionNode n;
56 n.id = intAttr(e, "id");
57 n.moduleName = e.attribute("name");
58 n.label = e.attribute("label", n.moduleName);
59 n.pos = { e.attribute("x").toDouble(), e.attribute("y").toDouble() };
60 n.enabled = intAttr(e, "enabled", 1) != 0;
61
62 QDomElement params = e.firstChildElement("params");
63 for (QDomElement p = params.firstChildElement("param");
64 !p.isNull(); p = p.nextSiblingElement("param"))
65 n.params[p.attribute("name")] = p.attribute("value");
66
67 QDomElement flags = e.firstChildElement("flags");
68 for (QDomElement fl = flags.firstChildElement("flag");
69 !fl.isNull(); fl = fl.nextSiblingElement("flag"))
70 n.flags.append(fl.attribute("name"));
71
72 mNextId = qMax(mNextId, n.id);
73 mActions.append(n);
74 }
75
76 for (QDomElement e = root.firstChildElement("data");
77 !e.isNull(); e = e.nextSiblingElement("data"))
78 {
79 ModelDataNode n;
80 n.id = intAttr(e, "id");
81 n.type = e.attribute("type", "raster");
82 n.value = e.attribute("value", {});
83 n.label = e.attribute("label", n.value);
84 n.pos = { e.attribute("x").toDouble(), e.attribute("y").toDouble() };
85 n.enabled = intAttr(e, "enabled", 1) != 0;
86 n.intermediate = intAttr(e, "intermediate", 0) != 0;
87
88 mNextId = qMax(mNextId, n.id);
89 mDataNodes.append(n);
90 }
91
92 for (QDomElement e = root.firstChildElement("relation");
93 !e.isNull(); e = e.nextSiblingElement("relation"))
94 {
95 ModelRelation r;
96 r.id = intAttr(e, "id");
97 r.fromId = intAttr(e, "from");
98 r.toId = intAttr(e, "to");
99 r.paramName = e.attribute("param");
100 mNextId = qMax(mNextId, r.id);
101 mRelations.append(r);
102 }
103
104 emit modelChanged();
105 return true;
106}

108bool GrassModel::saveFile(const QString &path) const
109{
110 QDomDocument doc;
111 QDomElement root = doc.createElement("model");
112 root.setAttribute("version", "0.9");
113 root.setAttribute("name", mName);
114 root.setAttribute("description", mDescription);
115 doc.appendChild(root);
116
117 for (const ModelActionNode &n : mActions)
118 {
119 QDomElement e = doc.createElement("action");
120 e.setAttribute("id", n.id);
121 e.setAttribute("name", n.moduleName);
122 e.setAttribute("label", n.label);
123 e.setAttribute("x", n.pos.x());
124 e.setAttribute("y", n.pos.y());
125 e.setAttribute("enabled", n.enabled ? 1 : 0);
126
127 if (!n.params.isEmpty())
128 {
129 QDomElement pe = doc.createElement("params");
130 for (auto it = n.params.cbegin(); it != n.params.cend(); ++it)
131 {
132 QDomElement p = doc.createElement("param");
133 p.setAttribute("name", it.key());
134 p.setAttribute("value", it.value());
135 pe.appendChild(p);
136 }
137 e.appendChild(pe);
138 }
139
140 if (!n.flags.isEmpty())
141 {
142 QDomElement fe = doc.createElement("flags");
143 for (const QString &f : n.flags)
144 {
145 QDomElement fl = doc.createElement("flag");
146 fl.setAttribute("name", f);
147 fe.appendChild(fl);
148 }
149 e.appendChild(fe);
150 }
151 root.appendChild(e);
152 }
153
154 for (const ModelDataNode &n : mDataNodes)
155 {
156 QDomElement e = doc.createElement("data");
157 e.setAttribute("id", n.id);
158 e.setAttribute("type", n.type);
159 e.setAttribute("value", n.value);
160 e.setAttribute("label", n.label);
161 e.setAttribute("x", n.pos.x());
162 e.setAttribute("y", n.pos.y());
163 e.setAttribute("enabled", n.enabled ? 1 : 0);
164 e.setAttribute("intermediate", n.intermediate ? 1 : 0);
165 root.appendChild(e);
166 }
167
168 for (const ModelRelation &r : mRelations)
169 {
170 QDomElement e = doc.createElement("relation");
171 e.setAttribute("id", r.id);
172 e.setAttribute("from", r.fromId);
173 e.setAttribute("to", r.toId);
174 e.setAttribute("param", r.paramName);
175 root.appendChild(e);
176 }
177
178 QFile f(path);
179 if (!f.open(QIODevice::WriteOnly | QIODevice::Text)) return false;
180 QTextStream out(&f);
181 doc.save(out, 2);
182 return true;
183}

227bool GrassModel::removeNode(int id)
228{
229 mRelations.erase(
230 std::remove_if(mRelations.begin(), mRelations.end(),
231 [id](const ModelRelation &r){ return r.fromId == id || r.toId == id; }),
232 mRelations.end());
233
234 int removed = 0;
235 mActions.erase(
236 std::remove_if(mActions.begin(), mActions.end(),
237 [id, &removed](const ModelActionNode &n){
238 if (n.id == id) { ++removed; return true; } return false; }),
239 mActions.end());
240
241 if (!removed)
242 mDataNodes.erase(
243 std::remove_if(mDataNodes.begin(), mDataNodes.end(),
244 [id, &removed](const ModelDataNode &n){
245 if (n.id == id) { ++removed; return true; } return false; }),
246 mDataNodes.end());
247
248 if (removed) { emit modelChanged(); return true; }
249 return false;
