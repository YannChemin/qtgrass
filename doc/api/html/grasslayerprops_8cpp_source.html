50static QString runModule( const QString &prog,
51 const QStringList &args )
52{
53 const QString gb = QString::fromLocal8Bit( qgetenv( "GISBASE" ) );
54 if ( gb.isEmpty() ) return "(GISBASE not set)";
55
56 QProcess p;
57 p.setProcessChannelMode( QProcess::SeparateChannels );
58 QProcessEnvironment env = GrassEnv::instance()->buildSubprocessEnv();
59 p.setProcessEnvironment( env );
60
61 const QString binary = gb + "/bin/" + prog;
62 qCInfo(lcLayerProps) << "runModule:" << binary << args;
63 p.start( binary, args );
64
65 if ( !p.waitForStarted( 5000 ) )
66 return QString("(failed to start %1: %2)").arg(prog).arg(p.errorString());
67
68 if ( !p.waitForFinished( 15000 ) )
69 {
70 p.kill();
71 return QString("(timeout running %1)").arg(prog);
72 }
73
74 const QString out = QString::fromLocal8Bit( p.readAllStandardOutput() );
75 const QString err = QString::fromLocal8Bit( p.readAllStandardError() );
76
77 if ( !err.isEmpty() )
78 qCWarning(lcLayerProps) << "runModule" << prog << "stderr:" << err.left(500);
79
80 if ( out.isEmpty() && !err.isEmpty() )
81 return err; // show the error text so the user knows what went wrong
82
83 return out;
84}

90// Add a themed group box with key-value rows from a JSON object.
91// `keys` is an ordered list of {jsonKey, humanLabel} pairs; null/empty values
92// are omitted so the group box collapses cleanly when data is absent.
93static QGroupBox *jsonGroup( const QString &title,
94 const QJsonObject &obj,
95 const QVector<QPair<QString,QString>> &keys,
96 QWidget *parent )
97{
98 auto *box = new QGroupBox( title, parent );
99 auto *form = new QFormLayout( box );
100 form->setLabelAlignment( Qt::AlignRight | Qt::AlignVCenter );
101 form->setContentsMargins( 8, 4, 8, 6 );
102
103 for ( const auto &[k, lbl] : keys )
104 {
105 const QJsonValue v = obj.value( k );
106 if ( v.isUndefined() || v.isNull() ) continue;
107 QString text;
108 if ( v.isBool() ) text = v.toBool() ? "yes" : "no";
109 else if ( v.isDouble() ) text = QString::number( v.toDouble(), 'g', 12 );
110 else text = v.toString().trimmed();
111 if ( text.isEmpty() ) continue;
112 auto *valLbl = new QLabel( text, box );
113 valLbl->setTextInteractionFlags( Qt::TextSelectableByMouse );
114 valLbl->setWordWrap( true );
115 form->addRow( "<b>" + lbl + "</b>", valLbl );
116 }
117 return box;
118}
