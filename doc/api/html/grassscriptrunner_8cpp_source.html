84void GrassScriptRunner::run()
85{
86 if ( mProcess && mProcess->state() != QProcess::NotRunning ) return;
87
88 mQueue.clear();
89 mLog->clear();
90
91 for ( const QString &line : mEditor->toPlainText().split( '\n', Qt::SkipEmptyParts ) )
92 {
93 const QString cmd = line.trimmed();
94 if ( cmd.isEmpty() || cmd.startsWith( '#' ) ) continue;
95 mQueue.append( cmd );
96 }
97
98 if ( mQueue.isEmpty() ) { mStatus->setText( "Nothing to run." ); return; }
99
100 mTotal = mQueue.size();
101 mDone = 0;
102 runNext();
103}

105bool GrassScriptRunner::runNext()
106{
107 if ( mQueue.isEmpty() )
108 {
109 mStatus->setText( QString( "✓ Finished — %1 command(s) run." ).arg( mTotal ) );
110 return false;
111 }
112
113 const QString cmd = mQueue.takeFirst();
114 ++mDone;
115 mStatus->setText( QString( "Running %1/%2: %3" ).arg( mDone ).arg( mTotal ).arg( cmd ) );
116 mLog->appendPlainText( QString( "\n$ %1" ).arg( cmd ) );
117
118 const QString gb = QString::fromLocal8Bit( qgetenv( "GISBASE" ) );
119
120 // Split command into argv (naive: split on spaces — fine for GRASS key=val style)
121 const QStringList parts = cmd.split( ' ', Qt::SkipEmptyParts );
122 if ( parts.isEmpty() ) { runNext(); return true; }
123
124 const QString prog = parts.first().contains( '/' )
125 ? parts.first()
126 : ( gb + "/bin/" + parts.first() );
127 const QStringList args = parts.mid( 1 );
128
129 QProcessEnvironment env = GrassEnv::instance()->buildSubprocessEnv();
130 const QString oldPath = env.value( "PATH" );
131 env.insert( "PATH", gb + "/bin:" + gb + "/scripts:" + oldPath );
132
133 mProcess = new QProcess( this );
134 mProcess->setProcessEnvironment( env );
135 mProcess->setProcessChannelMode( QProcess::MergedChannels );
136
137 connect( mProcess, &QProcess::readyReadStandardOutput,
138 this, &GrassScriptRunner::onProcessOutput );
139 connect( mProcess, qOverload<int, QProcess::ExitStatus>( &QProcess::finished ),
140 this, &GrassScriptRunner::onProcessFinished );
141
142 mProcess->start( prog, args );
143 return true;
144}
