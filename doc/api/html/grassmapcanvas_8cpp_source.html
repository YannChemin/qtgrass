107void GrassMapCanvas::zoomToLayerIfNeeded( const GrassExtent &layerExt )
108{
109 // Decide whether to zoom based on how visible the new layer is in the
110 // current viewport, regardless of layer count.
111 //
112 // Zoom if:
113 // - viewport is invalid (not yet initialised)
114 // - layer extent is invalid (load failed — defer)
115 // - layer does not overlap the viewport at all
116 // - layer covers < 1% of the viewport area (e.g. a 15 km raster in a
117 // 500 km NC state-wide view)
118 const GrassExtent vp = viewportExtent();
119
120 if ( !layerExt.isValid() )
121 {
122 // Extent not loaded yet — defer and retry
123 QTimer::singleShot( 80, this, [this] { zoomToFullExtent(); } );
124 return;
125 }
126
127 bool needZoom = !vp.isValid();
128
129 if ( !needZoom )
130 {
131 const double olN = std::min( layerExt.north, vp.north );
132 const double olS = std::max( layerExt.south, vp.south );
133 const double olE = std::min( layerExt.east, vp.east );
134 const double olW = std::max( layerExt.west, vp.west );
135
136 if ( olN <= olS || olE <= olW )
137 {
138 needZoom = true; // no overlap at all
139 }
140 else
141 {
142 // Overlap exists — zoom if layer is a tiny fraction of the viewport
143 const double overlapFrac = ( (olN-olS) * (olE-olW) )
144 / ( vp.height() * vp.width() );
145 needZoom = ( overlapFrac < 0.01 );
146 }
147 }
148
149 if ( needZoom )
150 {
151 zoomToFullExtent();
152 QTimer::singleShot( 100, this, [this] { zoomToFullExtent(); } );
153 }
154 else
155 {
156 scheduleRender();
157 }
158}

201void GrassMapCanvas::zoomToFullExtent()
202{
203 // Guard: if the canvas hasn't been laid out yet (width/height still 0),
204 // defer until resizeEvent gives us real dimensions. Computing
205 // mPixelsPerUnit with W=1 or H=1 produces a near-zero scale that makes
206 // the raster invisible until the user manually zooms.
207 if ( width() <= 0 || height() <= 0 )
208 {
209 QTimer::singleShot( 50, this, [this] { zoomToFullExtent(); } );
210 return;
211 }
212
213 // Determine whether the current GRASS session is geographic (degrees) or
214 // projected (metres). Use the cached region north value as a proxy: if
215 // |north| ≤ 90 the session is likely geographic, otherwise projected.
216 // This guards against mixing a global basemap (±180°/±85°) with projected
217 // rasters (N > 100 000 m) in the combined extent, which would produce a
218 // nonsensically huge viewport.
219 const bool sessionIsGeo = mRegion.valid &&
220 std::abs( mRegion.north ) <= 90.0 &&
221 std::abs( mRegion.east ) <= 360.0;
222
223 GrassExtent combined;
224 bool any = false;
225 for ( const GrassMapLayer *l : mLayers )
226 {
227 if ( !l->visible() ) continue;
228 const GrassExtent e = l->extent();
229 if ( !e.isValid() ) continue;
230
231 // Skip layers whose extent scale doesn't match the session projection.
232 const bool extIsGeo = std::abs( e.north ) <= 90.0 &&
233 std::abs( e.west ) <= 360.0;
234 if ( extIsGeo != sessionIsGeo ) continue;
235
236 if ( !any ) { combined = e; any = true; }
237 else
238 {
239 combined.north = std::max( combined.north, e.north );
240 combined.south = std::min( combined.south, e.south );
241 combined.east = std::max( combined.east, e.east );
242 combined.west = std::min( combined.west, e.west );
243 }
244 }
245 if ( !any || !combined.isValid() ) { scheduleRender(); return; }
246 const int W = width();
247 const int H = height();
248 mPixelsPerUnit = std::min( static_cast<double>(W) / ( combined.width() * 1.05 ),
249 static_cast<double>(H) / ( combined.height() * 1.05 ) );
250 mCenter = { ( combined.east + combined.west ) / 2.0,
251 ( combined.north + combined.south ) / 2.0 };
252 scheduleRender();
253}

327QString GrassMapCanvas::identifyRaster( GrassRasterMapLayer *layer,
328 double east, double north ) const
329{
330 const GrassExtent e = layer->extent();
331 if ( !e.isValid() || east < e.west || east > e.east ||
332 north > e.north || north < e.south )
333 return "outside extent";
334
335 const QByteArray nameBytes = layer->name().toLocal8Bit();
336 const QByteArray mapsetBytes = layer->mapset().toLocal8Bit();
337 const char *mapset = mapsetBytes.isEmpty() ? "" : mapsetBytes.constData();
338
339 QByteArray mutableName = nameBytes;
340 if ( !G_find_raster( mutableName.data(), mapset ) )
341 return "map not found";
342
343 struct Cell_head hd;
344 Rast_get_cellhd( nameBytes.constData(), mapset, &hd );
345
346 const int row = static_cast<int>( ( hd.north - north ) / hd.ns_res );
347 const int col = static_cast<int>( ( east - hd.west ) / hd.ew_res );
348 if ( row < 0 || row >= hd.rows || col < 0 || col >= hd.cols )
349 return "outside raster";
350
351 const int fd = Rast_open_old( nameBytes.constData(), mapset );
352 if ( fd < 0 ) return "cannot open";
353
354 const RASTER_MAP_TYPE type = Rast_get_map_type( fd );
355 void *buf = Rast_allocate_buf( type );
356
357 Rast_get_row( fd, buf, row, type );
358 Rast_close( fd );
359
360 QString result;
361 if ( Rast_is_null_value( reinterpret_cast<char *>( buf ) + col * Rast_cell_size( type ),
362 type ) )
363 {
364 result = "null";
365 }
366 else
367 {
368 switch ( type )
369 {
370 case CELL_TYPE:
371 {
372 const CELL v = reinterpret_cast<CELL *>( buf )[col];
373 result = QString::number( v );
374 break;
375 }
376 case FCELL_TYPE:
377 {
378 const FCELL v = reinterpret_cast<FCELL *>( buf )[col];
379 result = QString::number( static_cast<double>( v ), 'g', 8 );
380 break;
381 }
382 case DCELL_TYPE:
383 {
384 const DCELL v = reinterpret_cast<DCELL *>( buf )[col];
385 result = QString::number( v, 'g', 12 );
386 break;
387 }
388 }
389 }
390
391 G_free( buf );
392 return result;
393}

466QList<GrassDisplayCmd> GrassMapCanvas::buildDisplayCmds() const
467{
468 QList<GrassDisplayCmd> cmds;
469 for ( GrassMapLayer *layer : mLayers )
470 {
471 if ( !layer->visible() ) continue;
472
473 if ( auto *rl = dynamic_cast<GrassRasterMapLayer *>( layer ) )
474 {
475 const QString ref = rl->mapset().isEmpty()
476 ? rl->name()
477 : rl->name() + "@" + rl->mapset();
478 cmds.append( GrassDisplayCmd::raster( ref ) );
479 }
480 else if ( auto *vl = dynamic_cast<GrassVectorMapLayer *>( layer ) )
481 {
482 const QString ref = vl->mapset().isEmpty()
483 ? vl->name()
484 : vl->name() + "@" + vl->mapset();
485 cmds.append( GrassDisplayCmd::vector( ref,
486 vl->style().lineColor, vl->style().fillColor,
487 static_cast<int>( vl->style().lineWidth ) ) );
488 }
489 }
490 cmds << mDecorations;
491 return cmds;
492}

774void GrassMapCanvas::startAsyncRender()
775{
776 if ( mRenderMode != RenderMode::Direct ) return;
777
778 // If a render is already running, just flag that another one is needed.
779 if ( mFutureWatcher->isRunning() )
780 {
781 mRenderPending = true;
782 return;
783 }
784
785 const int W = width();
786 const int H = height();
787 if ( W <= 0 || H <= 0 ) return;
788
789 const GrassExtent vp = viewportExtent();
790
791 // Capture everything the render lambda needs by value.
792 // Raw layer pointers are safe: layers outlive the render (clearLayers waits).
793 QList<GrassMapLayer *> layers = mLayers;
794 const bool isEmpty = mLayers.isEmpty();
795 const RegionCache region = mRegion;
796 const bool showRegion = mShowRegion;
797 const QList<GrassDisplayCmd> decos = mDecorations;
798 // mRenderCenter / mRenderPpu are updated in onRenderFinished.
799
800 emit renderingStarted();
801
802 QFuture<QImage> future = QtConcurrent::run( [=]() -> QImage
803 {
804 QImage canvas( W, H, QImage::Format_ARGB32_Premultiplied );
805 canvas.fill( QColor( 26, 26, 26 ) ); // opaque dark background
806 QPainter p( &canvas );
807
808 // "No layers" placeholder and attribution text are drawn by paintEvent()
809 // on the main thread — QFont/drawText are not safe in worker threads.
810 if ( !isEmpty )
811 {
812 for ( GrassMapLayer *layer : layers )
813 {
814 if ( !layer->visible() ) continue;
815 const QImage img = layer->render( W, H, vp );
816 if ( img.isNull() ) continue;
817 p.setOpacity( layer->opacity() );
818 p.drawImage( 0, 0, img );
819 p.setOpacity( 1.0 );
820 }
821 }
822
823 // Region overlay — geometry only; label drawn by paintEvent (QFont unsafe here)
824 if ( showRegion && region.valid )
825 {
826 auto vp2px = [&]( double mx, double my ) -> QPointF {
827 return { ( mx - vp.west ) / vp.width() * W,
828 ( vp.north - my ) / vp.height() * H };
829 };
830 const QPointF nw = vp2px( region.west, region.north );
831 const QPointF se = vp2px( region.east, region.south );
832 const QRectF rect( nw, se );
833 if ( !rect.isEmpty() )
834 {
835 p.setPen( QPen( QColor( 255, 200, 0, 200 ), 2.0, Qt::DashLine ) );
836 p.setBrush( QColor( 255, 200, 0, 20 ) );
837 p.drawRect( rect );
838 }
839 }
840
841 // ── Decoration overlays — drawn while painter is still open ───────
842 if ( !decos.isEmpty() )
843 drawDecorationsNative( p, W, H, vp, decos );
844
845 p.end();
846
847 return canvas; // QImage is thread-safe; QPixmap::fromImage() on main thread
848 } );
849
850 mFutureWatcher->setFuture( future );
851}

859 update();
860 emit renderingDone();
