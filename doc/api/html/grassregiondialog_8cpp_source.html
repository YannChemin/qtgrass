154void GrassRegionDialog::loadRegion()
155{
156 const QString gb = QString::fromLocal8Bit( qgetenv( "GISBASE" ) );
157 if ( gb.isEmpty() ) return;
158
159 QProcess p;
160 p.setProcessEnvironment( QProcessEnvironment::systemEnvironment() );
161 p.start( gb + "/bin/g.region", { "-p" } );
162 p.waitForFinished( 5000 );
163
164 const QString out = QString::fromLocal8Bit( p.readAllStandardOutput() );
165 QMap<QString, double> vals;
166 for ( const QString &line : out.split( '\n', Qt::SkipEmptyParts ) )
167 {
168 const int col = line.indexOf( ':' );
169 if ( col < 0 ) continue;
170 const QString key = line.left( col ).trimmed();
171 bool ok;
172 const double val = line.mid( col + 1 ).trimmed().toDouble( &ok );
173 if ( ok ) vals[key] = val;
174 }
175
176 mUpdating = true;
177 if ( vals.contains("north") ) mNorth->setValue( vals["north"] );
178 if ( vals.contains("south") ) mSouth->setValue( vals["south"] );
179 if ( vals.contains("east") ) mEast->setValue( vals["east"] );
180 if ( vals.contains("west") ) mWest->setValue( vals["west"] );
181 if ( vals.contains("nsres") ) mNSRes->setValue( vals["nsres"] );
182 if ( vals.contains("ewres") ) mEWRes->setValue( vals["ewres"] );
183 if ( vals.contains("rows") ) mRows->setValue( static_cast<int>( vals["rows"] ) );
184 if ( vals.contains("cols") ) mCols->setValue( static_cast<int>( vals["cols"] ) );
185 mUpdating = false;
186
187 mStatus->setStyleSheet( "color:#555; font-style:italic;" );
188 mStatus->setText( "Current computational region loaded." );
189}

194void GrassRegionDialog::nsresChanged()
195{
196 if ( mUpdating ) return;
197 const double ns = mNSRes->value();
198 const double ew = mEWRes->value();
199 if ( ns <= 0 || ew <= 0 ) return;
200 const double height = mNorth->value() - mSouth->value();
201 const double width = mEast->value() - mWest->value();
202 mUpdating = true;
203 mRows->setValue( std::max( 1, static_cast<int>( height / ns ) ) );
204 mCols->setValue( std::max( 1, static_cast<int>( width / ew ) ) );
205 mUpdating = false;
206}
