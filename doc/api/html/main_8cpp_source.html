28// ── Crash signal handler ──────────────────────────────────────────────────────
29// Writes a breadcrumb to stderr + log file, then re-raises with the default
30// handler so that the kernel writes a core dump for gdb bt.
31static void crashHandler( int sig )
32{
33 // async-signal-safe: write() only
34 const char *hdr = "\n!!! CRASH: signal ";
35 write( STDERR_FILENO, hdr, strlen(hdr) );
36 char buf[8]; int n = sig;
37 int len = 0;
38 do { buf[len++] = '0' + n % 10; n /= 10; } while (n);
39 for (int i = len-1; i >= 0; --i) write(STDERR_FILENO, &buf[i], 1);
40 const char *trailer = " — see ~/qtgrass-debug.log and core dump\n";
41 write( STDERR_FILENO, trailer, strlen(trailer) );
42
43 // Restore default + re-raise so the kernel writes a core dump
44 ::signal( sig, SIG_DFL );
45 ::raise( sig );
46}
