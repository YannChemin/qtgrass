66// ── Geographic (lat/lon) rendering — same coordinate system as tiles ───────
67static void renderGeo( QPainter &p, GrassTileCache *cache, GrassTileFetcher *fetcher,
68 const GrassExtent &vp, int viewW, int viewH )
69{
70 const double north = std::min( vp.north, TileCoords::kMaxLat );
71 const double south = std::max( vp.south, TileCoords::kMinLat );
72 const double east = std::min( vp.east, 180.0 );
73 const double west = std::max( vp.west, -180.0 );
74 if ( north <= south || east <= west ) return;
75
76 const int z = TileCoords::bestZoom( east-west, viewW, 0, 19 );
77 const int maxTile = (1<<z) - 1;
78 const int tx0 = std::max( 0, TileCoords::lonToTileX( west, z ) );
79 const int tx1 = std::min( maxTile, TileCoords::lonToTileX( east, z ) );
80 const int ty0 = std::max( 0, TileCoords::latToTileY( north, z ) );
81 const int ty1 = std::min( maxTile, TileCoords::latToTileY( south, z ) );
82
83 for ( int ty = ty0; ty <= ty1; ++ty )
84 for ( int tx = tx0; tx <= tx1; ++tx )
85 {
86 const GrassTileKey key{ z, tx, ty };
87 const QImage tile = cache->get( key );
88 if ( tile.isNull() ) { queueFetch(fetcher,z,tx,ty); continue; }
89
90 const double tW = TileCoords::tileXToLon( tx, z );
91 const double tE = TileCoords::tileXToLon( tx+1, z );
92 const double tN = TileCoords::tileYToLat( ty, z );
93 const double tS = TileCoords::tileYToLat( ty+1, z );
94
95 const int px = static_cast<int>(std::round((tW-vp.west)/vp.width()*viewW));
96 const int py = static_cast<int>(std::round((vp.north-tN)/vp.height()*viewH));
97 const int pw = static_cast<int>(std::round((tE-tW)/vp.width()*viewW));
98 const int ph = static_cast<int>(std::round((tN-tS)/vp.height()*viewH));
99 if ( pw > 0 && ph > 0 )
100 p.drawImage( QRect(px,py,pw,ph), tile );
101 }
102}

104// ── Projected CRS rendering — reproject tiles to the viewport's CRS ────────
105// Uses PROJ 6+ per-thread context for thread safety.
106// Each tile is warped using a perspective quad-to-quad transform.
107static void renderProjected( QPainter &p, GrassTileCache *cache, GrassTileFetcher *fetcher,
108 const GrassExtent &vp, int viewW, int viewH,
109 const QByteArray &fromWkt )
110{
111 if ( fromWkt.isEmpty() ) return;
112
113 // ── Set up thread-local PROJ transform: current CRS → WGS84 ───────────
114 PJ_CONTEXT *ctx = proj_context_create();
115 // PROJ can use WKT, EPSG codes, or PROJ strings as source definition
116 PJ *toWgs = proj_create_crs_to_crs( ctx,
117 fromWkt.constData(), "EPSG:4326", nullptr );
118 if ( !toWgs ) { proj_context_destroy(ctx); return; }
119 // Normalize so lon is always X, lat is always Y
120 PJ *norm = proj_normalize_for_visualization( ctx, toWgs );
121 proj_destroy( toWgs );
122 if ( !norm ) { proj_context_destroy(ctx); return; }
123
124 // Project projected (e,n) → WGS84 (lon,lat)
125 auto toLatLon = [&]( double e, double n, double &lon, double &lat )
126 {
127 PJ_COORD c = proj_coord( e, n, 0, 0 );
128 PJ_COORD r = proj_trans( norm, PJ_FWD, c );
129 lon = r.xy.x; lat = r.xy.y;
130 };
131 // Project WGS84 (lon,lat) → projected (e,n)
132 auto toProj = [&]( double lon, double lat, double &e, double &n )
133 {
134 PJ_COORD c = proj_coord( lon, lat, 0, 0 );
135 PJ_COORD r = proj_trans( norm, PJ_INV, c );
136 e = r.xy.x; n = r.xy.y;
137 };
138
139 // ── Get viewport bounding box in WGS84 ────────────────────────────────
140 double lon_nw, lat_nw, lon_ne, lat_ne, lon_sw, lat_sw, lon_se, lat_se;
141 toLatLon( vp.west, vp.north, lon_nw, lat_nw );
142 toLatLon( vp.east, vp.north, lon_ne, lat_ne );
143 toLatLon( vp.west, vp.south, lon_sw, lat_sw );
144 toLatLon( vp.east, vp.south, lon_se, lat_se );
145
146 const double lat_min = std::min({lat_nw,lat_ne,lat_sw,lat_se});
147 const double lat_max = std::max({lat_nw,lat_ne,lat_sw,lat_se});
148 const double lon_min = std::min({lon_nw,lon_ne,lon_sw,lon_se});
149 const double lon_max = std::max({lon_nw,lon_ne,lon_sw,lon_se});
150
151 if ( lat_max < TileCoords::kMinLat || lat_min > TileCoords::kMaxLat )
152 {
153 proj_destroy(norm); proj_context_destroy(ctx);
154 return;
155 }
156
157 // ── Select zoom and tile range ─────────────────────────────────────────
158 const int z = TileCoords::bestZoom( lon_max - lon_min, viewW, 0, 19 );
159 const int maxTile = (1<<z) - 1;
160 const int tx0 = std::max( 0, TileCoords::lonToTileX(
161 std::max(lon_min, -180.0), z ) );
162 const int tx1 = std::min( maxTile, TileCoords::lonToTileX(
163 std::min(lon_max, 180.0), z ) );
164 const int ty0 = std::max( 0, TileCoords::latToTileY(
165 std::min(lat_max, TileCoords::kMaxLat), z ) );
166 const int ty1 = std::min( maxTile, TileCoords::latToTileY(
167 std::max(lat_min, TileCoords::kMinLat), z ) );
168
169 // Helper: map projected (e,n) → viewport pixel
170 auto toVpPx = [&]( double e, double n ) -> QPointF {
171 return { (e - vp.west) / vp.width() * viewW,
172 (vp.north - n) / vp.height() * viewH };
173 };
174
175 p.setRenderHint( QPainter::SmoothPixmapTransform );
176
177 for ( int ty = ty0; ty <= ty1; ++ty )
178 for ( int tx = tx0; tx <= tx1; ++tx )
179 {
180 const GrassTileKey key{ z, tx, ty };
181 const QImage tile = cache->get( key );
182 if ( tile.isNull() ) { queueFetch(fetcher,z,tx,ty); continue; }
183
184 // Tile corners in WGS84
185 const double tLon0 = TileCoords::tileXToLon( tx, z );
186 const double tLon1 = TileCoords::tileXToLon( tx+1, z );
187 const double tLat0 = TileCoords::tileYToLat( ty, z ); // north
188 const double tLat1 = TileCoords::tileYToLat( ty+1, z ); // south
189
190 // Tile corners → projected
191 double e00,n00, e10,n10, e01,n01, e11,n11;
192 toProj( tLon0, tLat0, e00, n00 ); // NW
193 toProj( tLon1, tLat0, e10, n10 ); // NE
194 toProj( tLon0, tLat1, e01, n01 ); // SW
195 toProj( tLon1, tLat1, e11, n11 ); // SE
196
197 // Projected → viewport pixels
198 const QPointF pNW = toVpPx( e00, n00 );
199 const QPointF pNE = toVpPx( e10, n10 );
200 const QPointF pSW = toVpPx( e01, n01 );
201 const QPointF pSE = toVpPx( e11, n11 );
202
203 const int tw = tile.width();
204 const int th = tile.height();
205
206 // Warp tile quad → viewport quad using a perspective transform
207 QPolygonF src, dst;
208 src << QPointF(0,0) << QPointF(tw,0) << QPointF(tw,th) << QPointF(0,th);
209 dst << pNW << pNE << pSE << pSW;
210
211 QTransform xf;
212 if ( QTransform::quadToQuad( src, dst, xf ) )
213 {
214 p.setTransform( xf );
215 p.drawImage( QPointF(0,0), tile );
216 p.resetTransform();
217 }
218 }
219
220 proj_destroy( norm );
221 proj_context_destroy( ctx );
222}

224// ── Public render entry point ──────────────────────────────────────────────
225QImage GrassTileBackground::render( int viewW, int viewH,
226 const GrassExtent &vp ) const
227{
228 if ( cacheHit( viewW, viewH, vp ) )
229 return cachedImage();
230
231 QImage img( viewW, viewH, QImage::Format_RGB32 );
232 img.fill( QColor( 30, 30, 30 ) ); // dark ocean fallback
233
234 if ( viewW <= 0 || viewH <= 0 || !vp.isValid() )
235 {
236 storeCache( viewW, viewH, vp, img );
237 return img;
238 }
239
240 QPainter p( &img );
241
242 if ( mIsGeographic )
243 renderGeo( p, mCache, mFetcher, vp, viewW, viewH );
244 else if ( !mProjFromWkt.isEmpty() )
245 renderProjected( p, mCache, mFetcher, vp, viewW, viewH, mProjFromWkt );
246 // else: no projection info → stays dark grey
247
248 p.end();
249 storeCache( viewW, viewH, vp, img );
250 return img;
251}
