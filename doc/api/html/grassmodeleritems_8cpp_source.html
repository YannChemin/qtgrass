270 const QStyleOptionGraphicsItem *opt, QWidget *)
271{
272 p->setRenderHint(QPainter::Antialiasing);
273
274 const bool sel = opt->state & QStyle::State_Selected;
275 QColor col = sel ? QColor(200, 140, 0) : QColor(80, 80, 140);
276
277 p->setPen(QPen(col, sel ? 2.5 : 1.8));
278 p->setBrush(Qt::NoBrush);
279 p->drawPath(mPath);
280
281 // Arrowhead at mP2
282 const double dx = qAbs(mP2.x() - mP1.x()) * 0.55 + 30.0;
283 const QPointF ctrl2 = mP2 - QPointF(dx, 0);
284 const double angle = qAtan2(mP2.y() - ctrl2.y(), mP2.x() - ctrl2.x());
285
286 static constexpr double L = 10.0, W2 = 5.0;
287 QPolygonF head;
288 head << mP2
289 << QPointF(mP2.x() - L * qCos(angle) + W2 * qSin(angle),
290 mP2.y() - L * qSin(angle) - W2 * qCos(angle))
291 << QPointF(mP2.x() - L * qCos(angle) - W2 * qSin(angle),
292 mP2.y() - L * qSin(angle) + W2 * qCos(angle));
293
294 p->setBrush(col);
295 p->setPen(Qt::NoPen);
296 p->drawPolygon(head);
297
298 // Param label at midpoint
299 if (!mParam.isEmpty())

303 f.setPointSize(7);
