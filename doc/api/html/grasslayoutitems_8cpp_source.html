101QRectF GrassLayoutItem::handleRect(Handle h) const
102{
103 const double s = kHS;
104 const double cx = mW / 2, cy = mH / 2;
105 switch (h)
106 {
107 case TL: return QRectF( 0 - s/2, 0 - s/2, s, s);
108 case TM: return QRectF(cx - s/2, 0 - s/2, s, s);
109 case TR: return QRectF(mW - s/2, 0 - s/2, s, s);
110 case ML: return QRectF( 0 - s/2, cy - s/2, s, s);
111 case MR: return QRectF(mW - s/2, cy - s/2, s, s);
112 case BL: return QRectF( 0 - s/2, mH - s/2, s, s);
113 case BM: return QRectF(cx - s/2, mH - s/2, s, s);
114 case BR: return QRectF(mW - s/2, mH - s/2, s, s);
115 default: return {};
116 }
117}

127void GrassLayoutItem::applyResize(Handle h, const QPointF &delta)
128{
129 double newW = mWAtDrag, newH = mHAtDrag;
130 QPointF newPos = mPosAtDrag;
131
132 switch (h)
133 {
134 case TL: newPos.rx() += delta.x(); newPos.ry() += delta.y();
135 newW -= delta.x(); newH -= delta.y(); break;
136 case TM: newPos.ry() += delta.y(); newH -= delta.y(); break;
137 case TR: newW += delta.x(); newPos.ry() += delta.y(); newH -= delta.y(); break;
138 case ML: newPos.rx() += delta.x(); newW -= delta.x(); break;
139 case MR: newW += delta.x(); break;
140 case BL: newPos.rx() += delta.x(); newW -= delta.x(); newH += delta.y(); break;
141 case BM: newH += delta.y(); break;
142 case BR: newW += delta.x(); newH += delta.y(); break;
143 default: break;
144 }
145
146 newW = qMax(10.0, newW);
147 newH = qMax(5.0, newH);
148
149 prepareGeometryChange();
150 setPos(newPos);
151 mW = newW;
152 mH = newH;
153 update();
154 emit geometryChanged();
155}

331double GrassLayoutScaleBarItem::niceNumber(double x)
332{
333 if (x <= 0) return 1;
334 const double p = std::pow(10, std::floor(std::log10(x)));
335 const double f = x / p;
336 double nice = 10;
337 if (f < 1.5) nice = 1;
338 else if (f < 3.0) nice = 2;
339 else if (f < 7.0) nice = 5;
340 return nice * p;
341}

357void GrassLayoutScaleBarItem::paintContent(QPainter *p, const QRectF &r)
358{
359 p->setRenderHint(QPainter::Antialiasing, false);
360
361 // Compute scale ratio (map units per mm)
362 double scaleUperMM = 1.0;
363 if (mCanvas && mMapFrame)
364 {
365 const auto ext = mCanvas->viewportExtent();
366 const double geoW = ext.width();
367 const double frameW = mMapFrame->itemW(); // mm
368 if (frameW > 0) scaleUperMM = geoW / frameW;
369 }
370
371 // Target bar length: 80% of item width
372 const double targetBarMM = r.width() * 0.8;
373 const double targetBarGeo = targetBarMM * scaleUperMM;
374 const double niceGeo = niceNumber(targetBarGeo);
375 const double barMM = niceGeo / scaleUperMM;
376
377 const double barH = r.height() * 0.4;
378 const double barY = r.center().y() - barH / 2;
379 const double barX = r.x() + (r.width() - barMM) / 2;
380
381 const int segs = qMax(1, mSegments);
382 const double segW = barMM / segs;
383
384 for (int i = 0; i < segs; ++i)
385 {
386 const QRectF segR(barX + i * segW, barY, segW, barH);
387 p->fillRect(segR, (i % 2 == 0) ? Qt::black : Qt::white);
388 p->setPen(QPen(Qt::black, 0.4));
389 p->drawRect(segR);
390 }
391
392 // Labels
393 QFont f; f.setPointSize(6); p->setFont(f); p->setPen(Qt::black);
394 p->drawText(QRectF(barX - 6, barY - 9, 12, 8), Qt::AlignCenter, "0");
395 const QString rightLabel = formatDist(niceGeo, mUnits);
396 p->drawText(QRectF(barX + barMM - 12, barY - 9, 24, 8),
397 Qt::AlignCenter, rightLabel);
398}

410{

440 outline.lineTo(ctr.x() + aw, base);

443 outline.closeSubpath();
444 p->drawPath(outline);

447 QFont f; f.setBold(true); f.setPointSize(7);
