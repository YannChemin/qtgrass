150void GrassMapExportDialog::doExport()
151{
152 const QString path = mPathEdit->text().trimmed();
153 if ( path.isEmpty() )
154 {
155 mStatus->setText( "Please choose an output file." );
156 return;
157 }
158
159 const int W = mWidthSpin->value();
160 const int H = mHeightSpin->value();
161
162 const int fmt = mFormatCombo->currentIndex();
163 bool ok = false;
164
165 if ( fmt == 3 ) { exportPdf( path, W, H ); ok = true; }
166 else if ( fmt == 4 ) { exportSvg( path, W, H ); ok = true; }
167 else { exportPng( path, W, H ); ok = true; }
168
169 if ( ok )
170 {
171 mStatus->setStyleSheet( "color:#1a8a1a; font-weight:bold;" );
172 mStatus->setText( QString( "Saved: %1 (%2 × %3 px)" )
173 .arg( path ).arg( W ).arg( H ) );
174 }
175}

180void GrassMapExportDialog::exportPng( const QString &path, int w, int h )
181{
182 const GrassExtent vp = mCanvas->viewportExtent();
183
184 // Render all visible layers at target resolution
185 QImage img( w, h, QImage::Format_RGB32 );
186 img.fill( QColor( 26, 26, 26 ) );
187 QPainter p( &img );
188
189 for ( GrassMapLayer *layer : mCanvas->layers() )
190 {
191 if ( !layer->visible() ) continue;
192 const QImage li = layer->render( w, h, vp );
193 if ( !li.isNull() ) p.drawImage( 0, 0, li );
194 }
195 p.end();
196
197 if ( !img.save( path ) )
198 mStatus->setText( "Failed to save image." );
199}
