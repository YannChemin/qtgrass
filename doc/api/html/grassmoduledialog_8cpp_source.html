144void GrassModuleDialog::buildOptionTabs( QTabWidget *tabs )
145{
146 // Determine which sections exist and create one tab per section.
147 // Params/flags with no guisection go into a "Required / Optional" tab.
148 QStringList sections = mDesc.sections();
149 if ( sections.isEmpty() ) sections << QString(); // at least one tab
150
151 // Always have an "Options" catch-all for params without a section
152 bool hasUnsectioned = false;
153 for ( const auto &p : mDesc.params )
154 if ( p.guiSection.isEmpty() ) { hasUnsectioned = true; break; }
155 for ( const auto &f : mDesc.flags )
156 if ( !f.hidden && f.guiSection.isEmpty() ) { hasUnsectioned = true; break; }
157
158 // Build section → scrollable form mapping
159 auto makeScrollForm = [this]( const QString &section ) -> QWidget *
160 {
161 QScrollArea *scroll = new QScrollArea();
162 scroll->setWidgetResizable( true );
163 scroll->setFrameShape( QFrame::NoFrame );
164
165 QWidget *container = new QWidget();
166 QVBoxLayout *lay = new QVBoxLayout( container );
167 lay->setSpacing( 4 );
168 lay->setContentsMargins( 8, 8, 8, 8 );
169
170 // Parameters for this section
171 for ( const auto &pd : mDesc.params )
172 {
173 if ( pd.guiSection != section ) continue;
174 GrassParamWidget *w = GrassParamWidget::create( pd, container );
175 mParamWidgets.append( w );
176 lay->addWidget( w );
177 connect( w, &GrassParamWidget::valueChanged,
178 this, &GrassModuleDialog::updateCommandPreview );
179 }
180
181 // Flags for this section (exclude internal ones)
182 bool addedFlagHeader = false;
183 for ( const auto &fd : mDesc.flags )
184 {
185 if ( fd.hidden || fd.guiSection != section ) continue;
186 if ( !addedFlagHeader )
187 {
188 QLabel *fhdr = new QLabel( "<i>Flags</i>", container );
189 fhdr->setStyleSheet( QString( "color: %1; margin-top: 6px;" )
190 .arg( qApp->palette().color( QPalette::PlaceholderText ).name() ) );
191 lay->addWidget( fhdr );
192 addedFlagHeader = true;
193 }
194 GrassFlagWidget *fw = new GrassFlagWidget( fd, container );
195 mFlagWidgets.append( fw );
196 lay->addWidget( fw );
197 connect( fw, &GrassFlagWidget::valueChanged,
198 this, &GrassModuleDialog::updateCommandPreview );
199 }
200
201 lay->addStretch();
202 scroll->setWidget( container );
203 return scroll;
204 };
205
206 if ( hasUnsectioned )
207 {
208 tabs->addTab( makeScrollForm( "" ), sections.isEmpty() ? "Options" : "Options" );
209 }
210
211 for ( const QString &sec : sections )
212 {
213 if ( sec.isEmpty() ) continue;
214 tabs->addTab( makeScrollForm( sec ), sec );
215 }
216}

282void GrassModuleDialog::applyPreArgs( const QStringList &preArgs )
283{
284 for ( const QString &arg : preArgs )
285 {
286 const int eq = arg.indexOf( '=' );
287 if ( eq < 0 ) continue;
288 const QString key = arg.left( eq ).trimmed();
289 const QString val = arg.mid( eq + 1 ).trimmed();
290
291 for ( GrassParamWidget *w : mParamWidgets )
292 {
293 if ( w->key() != key ) continue;
294 w->setValue( val );
295 break;
296 }

359void GrassModuleDialog::run()
360{
361 if ( !validateAll() )
362 {
363 QMessageBox::warning( this, mModuleName,
364 "Please fill in all required parameters (shown in red)." );
365 return;
366 }
367
368 const QString bin = moduleBin();
369 if ( bin.isEmpty() )
370 {
371 QMessageBox::critical( this, mModuleName,
372 "Module binary not found. Check GISBASE." );
373 return;
374 }
375
376 // Switch to Output tab
377 mTabs->setCurrentWidget( mOutputView->parentWidget() );
378 // Actually find the tab index
379 for ( int i = 0; i < mTabs->count(); ++i )
380 if ( mTabs->tabText( i ) == "Output" )
381 mTabs->setCurrentIndex( i );
382
383 mOutputView->clear();
384 mOutputView->appendPlainText( "$ " + mCommandView->toPlainText() + "\n" );
385
386 mProcess = new QProcess( this );
387 mProcess->setProcessEnvironment( buildEnv() );
388 mProcess->setProcessChannelMode( QProcess::MergedChannels );
389
390 connect( mProcess, &QProcess::started,
391 this, &GrassModuleDialog::onProcessStarted );
392 connect( mProcess, &QProcess::readyReadStandardOutput,
393 this, &GrassModuleDialog::onProcessOutput );
394 connect( mProcess, qOverload<int, QProcess::ExitStatus>( &QProcess::finished ),
395 this, &GrassModuleDialog::onProcessFinished );
396
397 mProcess->start( bin, collectArgs() );
398}

427void GrassModuleDialog::onProcessFinished( int exitCode, int exitStatus )
428{
429 mProgress->setVisible( false );
430 mRunBtn->setEnabled( true );
431 mCancelBtn->setEnabled( false );
432
433 const bool success = ( exitStatus == QProcess::NormalExit && exitCode == 0 );
434 mOutputView->appendPlainText(
435 success ? "\n✓ Finished successfully."
436 : QString( "\n✗ Finished with exit code %1." ).arg( exitCode ) );
437
438 // Notify interface to add output layers (Part 4 will wire this up)
439 if ( success && mIface )
440 {
441 // Iterate over "new" (output) raster/vector params that were set
442 for ( const GrassParamWidget *w : mParamWidgets )
443 {
444 if ( !w->desc().hasGisprompt ) continue;
445 const auto &gp = w->desc().gisprompt;
446 if ( gp.age != "new" ) continue;
447 const QStringList vals = w->values();
448 if ( vals.isEmpty() ) continue;
449
450 const QString mapName = vals.first();
451 const QString loc = QString::fromLocal8Bit( qgetenv( "LOCATION_NAME" ) );
452 const QString ms = QString::fromLocal8Bit( qgetenv( "MAPSET" ) );
453 const QString gisdb= QString::fromLocal8Bit( qgetenv( "GISDBASE" ) );
454
455 if ( gp.prompt == "raster" )
456 {
457 const QString uri = gisdb + "/" + loc + "/" + ms + "/cell/" + mapName;
458 mIface->addRasterLayer( uri, mapName, "grassraster" );
459 }
460 else if ( gp.prompt == "vector" )
461 {
462 const QString uri = gisdb + "/" + loc + "/" + ms + "/vector/" + mapName;
463 mIface->addVectorLayer( uri, mapName, "grass" );
464 }
465 }
466 }
467
468 mProcess->deleteLater();
469 mProcess = nullptr;
