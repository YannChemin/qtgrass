30bool GrassEnv::init()
31{
32 mValid = false;
33 mError.clear();
34
35 // Determine GISBASE: env var takes priority, then the compile-time default
36 // baked in at build time (QTGRASS_DEFAULT_GISBASE = GRASS_PREFIX).
37 const char *gb = getenv( "GISBASE" );
38 if ( !gb || !*gb )
39 {
40#ifdef QTGRASS_DEFAULT_GISBASE
41 const char *defGisbase = QTGRASS_DEFAULT_GISBASE;
42 if ( defGisbase && *defGisbase && QDir( QString::fromUtf8(defGisbase) ).exists() )
43 {
44 qputenv( "GISBASE", defGisbase );
45 gb = getenv( "GISBASE" );
46 qCInfo(lcEnv) << "GISBASE not set — using build-time default:" << defGisbase;
47 }
48#endif
49 if ( !gb || !*gb )
50 {
51 mError = "GISBASE is not set. "
52 "Run qtgrass from inside a GRASS session, or set GISBASE manually.";
53 return false;
54 }
55 }
56 mGisBase = QString::fromUtf8( gb );
57
58 // Verify the installed header version matches the runtime library.
59 // G_no_gisinit() will call G_fatal_error() if they differ, which by
60 // default calls exit(). We don't try to catch that here; it's the same
61 // behaviour QGIS uses.
62 G_set_gisrc_mode( G_GISRC_MODE_MEMORY );
63 G_no_gisinit();
64
65 // --- Discover the active GRASS session ---
66 //
67 // Priority 1: $GISRC file (set by "grass --exec" and interactive GRASS
68 // shells). GRASS stores GISDBASE/LOCATION_NAME/MAPSET there,
69 // NOT as individual env vars. Format: "KEY: VALUE\n".
70 if ( const char *gisrcPath = getenv( "GISRC" ) )
71 {
72 QFile f( QString::fromLocal8Bit( gisrcPath ) );
73 if ( f.open( QIODevice::ReadOnly | QIODevice::Text ) )
74 {
75 QTextStream in( &f );
76 while ( !in.atEnd() )
77 {
78 const QString line = in.readLine();
79 const int colon = line.indexOf( ':' );
80 if ( colon < 0 ) continue;
81 const QString key = line.left( colon ).trimmed();
82 const QString val = line.mid( colon + 1 ).trimmed();
83 if ( key == "GISDBASE" ) mGisDatabase = val;
84 else if ( key == "LOCATION_NAME" ) mLocation = val;
85 else if ( key == "MAPSET" ) mMapset = val;
86 }
87 }
88 }
89
90 // Priority 2: individual env vars (set by the --qtui hook in parser.c,
91 // or when the caller exports them explicitly). These override
92 // the GISRC file values when both are present.
93 if ( const char *v = getenv( "GISDBASE" ) ) if ( *v ) mGisDatabase = QString::fromUtf8( v );
94 if ( const char *v = getenv( "LOCATION_NAME" ) ) if ( *v ) mLocation = QString::fromUtf8( v );
95 if ( const char *v = getenv( "MAPSET" ) ) if ( *v ) mMapset = QString::fromUtf8( v );
96
97 // Push the discovered session into GRASS's in-memory state so any GRASS
98 // C API call (G_mapset, G_get_window, Rast_open_old …) works immediately.
99 // Without this, G_mapset() fires G_fatal_error("MAPSET is not set").
100 if ( !mGisDatabase.isEmpty() && !mLocation.isEmpty() && !mMapset.isEmpty() )
101 {
102 G_setenv_nogisrc( "GISDBASE", mGisDatabase.toUtf8().constData() );
103 G_setenv_nogisrc( "LOCATION_NAME", mLocation.toUtf8().constData() );
104 G_setenv_nogisrc( "MAPSET", mMapset.toUtf8().constData() );
105
106 // Mirror into the real OS environment so child processes (g.list
107 // in GrassDataCatalog, module dialogs, etc.) inherit the session
108 // even when there is no $GISRC file (e.g. desktop or .gxw launch).
109 qputenv( "GISDBASE", mGisDatabase.toUtf8() );
110 qputenv( "LOCATION_NAME", mLocation.toUtf8() );
111 qputenv( "MAPSET", mMapset.toUtf8() );
112 }
113
114 mValid = true;
115 qCInfo(lcEnv) << "GrassEnv: GISBASE=" << mGisBase
116 << "GISDBASE=" << mGisDatabase
117 << "LOCATION=" << mLocation
118 << "MAPSET=" << mMapset;
119 return true;
120}

122bool GrassEnv::openMapset( const QString &gisDatabase,
123 const QString &location,
124 const QString &mapset )
125{
126 // Validate paths exist
127 QDir dir( gisDatabase + "/" + location + "/" + mapset );
128 if ( !dir.exists() )
129 {
130 mError = QString( "Mapset not found: %1" ).arg( dir.absolutePath() );
131 return false;
132 }
133
134 // Update in-memory GRASS env vars (used by the GRASS C API in this process)
135 G_setenv_nogisrc( "GISDBASE", gisDatabase.toUtf8().constData() );
136 G_setenv_nogisrc( "LOCATION_NAME", location.toUtf8().constData() );
137 G_setenv_nogisrc( "MAPSET", mapset.toUtf8().constData() );
138
139 // Mirror session into the real OS environment so that child processes
140 // (g.list in the data catalog, module dialogs, etc.) inherit the session.
141 // G_setenv_nogisrc() only updates GRASS's in-memory GISRC — it does NOT
142 // affect the process environment that QProcess children read.
143 qputenv( "GISDBASE", gisDatabase.toUtf8() );
144 qputenv( "LOCATION_NAME", location.toUtf8() );
145 qputenv( "MAPSET", mapset.toUtf8() );
146
147 // Re-read the new mapset's WIND file and activate it as the computational
148 // region. G_get_window() caches the window in G__.window; G_unset_window()
149 // clears the cache so the next G_get_window() reads from the new location's
150 // WIND file rather than returning the stale previous session's region.
151 // Without this, G_projection() / savedRegion.proj stay at the old location's
152 // projection type, causing Rast_set_window() to validate UTM coords as lat/lon.
153 {
154 struct Cell_head window;
155 G_unset_window();
156 G_get_window( &window );
157 G_set_window( &window );
158 }
159
160 mGisDatabase = gisDatabase;
161 mLocation = location;
162 mMapset = mapset;
163
164 emit mapsetChanged( mGisDatabase, mLocation, mMapset );
165 return true;
166}

168QProcessEnvironment GrassEnv::buildSubprocessEnv() const
169{
170 QProcessEnvironment env = QProcessEnvironment::systemEnvironment();
171
172 // Write a temporary GISRC file so child processes can call G_gisinit().
173 // qtgrass uses G_GISRC_MODE_MEMORY (no GISRC file); subprocesses need the file.
174 if ( !mGisDatabase.isEmpty() && !mLocation.isEmpty() && !mMapset.isEmpty() )
175 {
176 const QString gisrcPath = QDir::tempPath()
177 + QString( "/qtgrass_%1.gisrc" ).arg( QCoreApplication::applicationPid() );
178 QFile f( gisrcPath );
179 if ( f.open( QIODevice::WriteOnly | QIODevice::Text ) )
180 {
181 QTextStream ts( &f );
182 ts << "GISDBASE: " << mGisDatabase << "\n";
183 ts << "LOCATION_NAME: " << mLocation << "\n";
184 ts << "MAPSET: " << mMapset << "\n";
185 ts << "GUI: text\n";
186 f.close();
187 env.insert( "GISRC", gisrcPath );
188 }
189 }
190
191 // Ensure GISBASE and library paths are propagated.
192 if ( !mGisBase.isEmpty() )
193 {
194 env.insert( "GISBASE", mGisBase );
195
196 const QString oldPath = env.value( "PATH" );
197 env.insert( "PATH", mGisBase + "/bin:" + mGisBase + "/scripts:" + oldPath );
198
199 const QString oldLd = env.value( "LD_LIBRARY_PATH" );
200 env.insert( "LD_LIBRARY_PATH",
201 mGisBase + "/lib" + ( oldLd.isEmpty() ? "" : ":" + oldLd ) );
202
203 // GRASS Python packages (grass.script, grass.pygrass, …) live in
204 // $GISBASE/etc/python. Without this, Python-based modules such as
205 // i.ai, r.learn.*, v.class.ml etc. fail with
206 // ModuleNotFoundError: No module named 'grass'
207 const QString grassPy = mGisBase + "/etc/python";
208 const QString oldPyPath = env.value( "PYTHONPATH" );
209 env.insert( "PYTHONPATH",
210 grassPy + ( oldPyPath.isEmpty() ? "" : ":" + oldPyPath ) );
211 }
212
213 // Mirror the session vars so any code reading env vars also works.
214 env.insert( "GISDBASE", mGisDatabase );
215 env.insert( "LOCATION_NAME", mLocation );
216 env.insert( "MAPSET", mMapset );
217
218 // Python for script modules
219 if ( env.value( "GRASS_PYTHON" ).isEmpty() )
220 {
221 const QString py = QStandardPaths::findExecutable( "python3" );
222 if ( !py.isEmpty() )
223 env.insert( "GRASS_PYTHON", py );
224 }
225
226 return env;
227}
