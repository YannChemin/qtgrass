39GrassScatterWidget::AxisRange GrassScatterWidget::computeRange(int bandIdx) const
40{
41 double lo = std::numeric_limits<double>::max();
42 double hi = -std::numeric_limits<double>::max();
43 for (const auto &cls : mClasses)
44 {
45 if (bandIdx >= cls.stats.size()) continue;
46 const auto &s = cls.stats[bandIdx];
47 if (s.count == 0) continue;
48 lo = qMin(lo, s.mean - 3 * s.stdDev);
49 hi = qMax(hi, s.mean + 3 * s.stdDev);
50 }
51 if (lo >= hi) { lo = 0; hi = 255; }
52 const double margin = (hi - lo) * 0.05;
53 return { lo - margin, hi + margin };
54}
