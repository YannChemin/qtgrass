115void GrassChartWidget::drawLineChart( QPainter &p, const QRect &pa ) const
116{
117 double xMin = mData.first().first, xMax = mData.last().first;
118 double yMin = mData.first().second, yMax = mData.first().second;
119 for ( const auto &kv : mData )
120 {
121 yMin = std::min( yMin, kv.second );
122 yMax = std::max( yMax, kv.second );
123 }
124 if ( xMax == xMin ) xMax = xMin + 1;
125 if ( yMax == yMin ) { yMin -= 1; yMax += 1; }
126
127 auto toPixel = [&]( double x, double y ) -> QPointF {
128 return {
129 pa.left() + ( x - xMin ) / ( xMax - xMin ) * pa.width(),
130 pa.bottom()- ( y - yMin ) / ( yMax - yMin ) * pa.height()
131 };
132 };
133
134 // Y-axis ticks
135 p.setPen( QColor( "#555" ) );
136 const int nTicks = 5;
137 for ( int i = 0; i <= nTicks; ++i )
138 {
139 double yVal = yMin + i * ( yMax - yMin ) / nTicks;
140 QPointF pt = toPixel( xMin, yVal );
141 p.drawLine( pt - QPointF( 4, 0 ), pt );
142 p.setPen( QColor( "#888" ) );
143 p.drawText( QRectF( pa.left() - 48, pt.y() - 8, 42, 16 ),
144 Qt::AlignRight | Qt::AlignVCenter,
145 QString::number( yVal, 'g', 4 ) );
146 p.setPen( QColor( "#333" ) );
147 p.drawLine( pt, { static_cast<double>( pa.right() ), pt.y() } );
148 p.setPen( QColor( "#555" ) );
149 }
150
151 // Line path
152 QPainterPath path;
153 bool first = true;
154 for ( const auto &kv : mData )
155 {
156 if ( kv.second != kv.second ) continue; // skip NaN (marked as *)
157 QPointF px = toPixel( kv.first, kv.second );
158 if ( first ) { path.moveTo( px ); first = false; }
159 else path.lineTo( px );
160 }
161
162 // Fill under line
163 if ( !path.isEmpty() )
164 {
165 QPainterPath fill = path;
166 fill.lineTo( toPixel( mData.last().first, yMin ) );
167 fill.lineTo( toPixel( mData.first().first, yMin ) );
168 fill.closeSubpath();
169 p.fillPath( fill, QColor( 100, 160, 240, 60 ) );
170 }
171
172 p.setPen( QPen( QColor( "#5aabff" ), 2 ) );
173 p.setBrush( Qt::NoBrush );
174 p.drawPath( path );
175
176 // X axis ticks (distance)
177 p.setPen( QColor( "#888" ) );
178 for ( int i = 0; i <= 5; ++i )
179 {
180 double xVal = xMin + i * ( xMax - xMin ) / 5.0;
181 QPointF pt = toPixel( xVal, yMin );
182 p.drawText( QRectF( pt.x() - 25, pa.bottom() + 6, 50, 14 ),
183 Qt::AlignCenter,
184 QString::number( xVal, 'f', 0 ) );
185 }
186}

191void GrassChartWidget::drawBarChart( QPainter &p, const QRect &pa ) const
192{
193 double yMax = 1;
194 for ( const auto &kv : mData ) yMax = std::max( yMax, kv.second );
195
196 const int n = mData.size();
197 if ( n == 0 ) return;
198
199 const double barW = pa.width() / static_cast<double>( n );
200
201 // Y ticks
202 p.setPen( QColor( "#888" ) );
203 for ( int i = 0; i <= 4; ++i )
204 {
205 double yVal = yMax * i / 4.0;
206 double yPx = pa.bottom() - yVal / yMax * pa.height();
207 p.drawText( QRectF( pa.left() - 48, yPx - 8, 42, 16 ),
208 Qt::AlignRight | Qt::AlignVCenter,
209 QString::number( static_cast<long long>( yVal ) ) );
210 p.setPen( QColor( "#333" ) );
211 p.drawLine( QPointF( pa.left(), yPx ), QPointF( pa.right(), yPx ) );
212 p.setPen( QColor( "#888" ) );
213 }
214
215 // ── Determine label step from font metrics ────────────────────────────────
216 // Use QFontMetrics to find the longest label, then compute how many fit
217 // horizontally without overlap, and rotate 45° to save vertical space.
218 QFont xFont( "sans", 7 );
219 QFontMetrics fm( xFont );
220 int maxLabelPx = 10;
221 for ( const QString &lbl : mBarLabels )
222 maxLabelPx = std::max( maxLabelPx, fm.horizontalAdvance( lbl ) );
223
224 // With 45° rotation each label needs maxLabelPx * cos(45°) ≈ 0.71 × px
225 // horizontal space. We require at least 1.2× that for breathing room.
226 const double labelSlotW = maxLabelPx * 0.71 * 1.2;
227
228 // How many labels can fit? Cap between 2 and 8.
229 const int maxLabels = std::min( 8, std::max( 2,
230 static_cast<int>( pa.width() / std::max( labelSlotW, 1.0 ) ) ) );
231
232 // Step between labelled bars
233 int step = 1;
234 if ( n > maxLabels )
235 step = ( n + maxLabels - 1 ) / maxLabels;
236
237 // ── Bars ─────────────────────────────────────────────────────────────────
238 for ( int i = 0; i < n; ++i )
239 {
240 const double h = mData[i].second / yMax * pa.height();
241 QRectF bar( pa.left() + i * barW + 1, pa.bottom() - h, barW - 2, h );
242 p.fillRect( bar, QColor( 80, 160, 220, 200 ) );
243 p.setPen( QColor( "#5aabff" ) );
244 p.drawRect( bar );
245 }
246
247 // ── X-axis labels (rotated 45°) ───────────────────────────────────────────
248 p.setFont( xFont );
249 for ( int i = 0; i < n; ++i )
250 {
251 // Always label first and last; otherwise every `step` bars
252 const bool isFirst = ( i == 0 );
253 const bool isLast = ( i == n - 1 );
254 if ( !isFirst && !isLast && ( i % step != 0 ) ) continue;
255 if ( i >= mBarLabels.size() ) continue;
256
257 // Shorten long labels: try to fit in at most 8 significant chars
258 QString lbl = mBarLabels[i];
259 bool ok;
260 const double val = lbl.toDouble( &ok );
261 if ( ok )
262 lbl = QString::number( val, 'g', 5 ); // ≤5 sig. figs.
263
264 // Tick mark
265 const double xCenter = pa.left() + ( i + 0.5 ) * barW;
266 p.setPen( QColor( "#555" ) );
267 p.drawLine( QPointF( xCenter, pa.bottom() ),
268 QPointF( xCenter, pa.bottom() + 4 ) );
269
270 // Rotated label: anchor at tick, text goes down-right at 45°
271 p.setPen( QColor( "#aaa" ) );
272 p.save();
273 p.translate( xCenter, pa.bottom() + 5 );
274 p.rotate( 45 );
275 p.drawText( QRect( 0, -fm.height() / 2, maxLabelPx + 10, fm.height() ),
276 Qt::AlignLeft | Qt::AlignVCenter, lbl );
277 p.restore();
278 }
