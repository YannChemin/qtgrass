27QPixmap GrassMapSwipeWidget::renderMap(const QString &map, int w, int h)
28{
29 if (map.isEmpty() || w < 1 || h < 1) return {};
30
31 QTemporaryFile tmp;
32 tmp.setFileTemplate(QDir::tempPath() + "/qtgrass_swipe_XXXXXX.png");
33 tmp.setAutoRemove(false);
34 if (!tmp.open()) return {};
35 const QString path = tmp.fileName();
36 tmp.close();
37
38 QProcessEnvironment env = GrassEnv::instance()->buildSubprocessEnv();
39 env.insert("GRASS_RENDER_IMMEDIATE", "cairo");
40 env.insert("GRASS_RENDER_FILE", path);
41 env.insert("GRASS_RENDER_FILE_READ", "FALSE");
42 env.insert("GRASS_RENDER_WIDTH", QString::number(w));
43 env.insert("GRASS_RENDER_HEIGHT", QString::number(h));
44 env.insert("GRASS_RENDER_BACKGROUNDCOLOR", "white");
45
46 QProcess p;
47 p.setProcessEnvironment(env);
48 p.setProcessChannelMode(QProcess::MergedChannels);
49 p.start("d.rast", {"map=" + map});
50 p.waitForFinished(30000);
51
52 QPixmap pix(path);
53 QFile::remove(path);
54 return pix;
55}
