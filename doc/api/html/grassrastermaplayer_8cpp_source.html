28{

36 QByteArray mutableName = nameBytes;
37 if ( !G_find_raster( mutableName.data(), mapset ) )
38 {
39 mError = QString( "Raster '%1' not found in mapset '%2'" )
40 .arg( mName, mapset[0] ? mapset : "(current)" );
41 qCWarning(lcCanvas) << "GrassRasterMapLayer:" << mError;
42 return false;
43 }
44
45 struct Cell_head hd;
46 Rast_get_cellhd( nameBytes.constData(), mapset, &hd );
47
48 // Detect rasters whose cellhd is internally inconsistent: the proj field
49 // says PROJECTION_LL (geographic) but the coordinates are in projected
50 // metres (e.g. a raster raw-copied from a UTM location into a lat/lon
51 // location without reprojection). Using such extents for zoomToFullExtent
52 // sets mCenter to millions of metres, which later causes
53 // G_fatal_error("Illegal latitude") in Rast_set_window.
54 // Use hd.proj (from the raster's own cellhd) — NOT G_projection(), which
55 // is frozen at the startup location and wrong after a session switch.
56 if ( hd.proj == PROJECTION_LL )
57 {
58 if ( std::abs( hd.north ) > 90.0 || std::abs( hd.south ) > 90.0
59 || std::abs( hd.east ) > 360.0 || std::abs( hd.west ) > 360.0 )
60 {
61 mError = QString( "Raster '%1' has projected extent in a geographic location"
62 " (north=%2) — skipping" ).arg( mName ).arg( hd.north );
63 qCWarning(lcCanvas) << "GrassRasterMapLayer:" << mError;
64 return false;
65 }
66 }
67
68 mExtent = { hd.north, hd.south, hd.east, hd.west };
69 mRows = hd.rows;
70 mCols = hd.cols;
71 mNsRes = hd.ns_res;
72 mEwRes = hd.ew_res;
73 return true;

84{

88
89 const QString db = QString::fromLocal8Bit( qgetenv( "GISDBASE" ) );
90 const QString loc = QString::fromLocal8Bit( qgetenv( "LOCATION_NAME" ) );
91 const QString ms = mMapset.isEmpty()
92 ? QString::fromLocal8Bit( qgetenv( "MAPSET" ) )
93 : mMapset;
94
95 if ( db.isEmpty() || loc.isEmpty() || ms.isEmpty() ) return;
96
97 const QString markerPath = QString( "%1/%2/%3/cell_misc/%4/qtgrass_ovr" )
98 .arg( db, loc, ms, mName );
99 QFile f( markerPath );
100 if ( !f.open( QIODevice::ReadOnly | QIODevice::Text ) ) return;
101
102 QTextStream ts( &f );
103 while ( !ts.atEnd() )
104 {
105 const QString line = ts.readLine().trimmed();
106 if ( line.isEmpty() || line.startsWith( '#' ) ) continue;
107 const QStringList parts = line.split( ' ', Qt::SkipEmptyParts );
108 if ( parts.size() < 2 ) continue;
109 bool ok;
110 const int factor = parts[0].toInt( &ok );
111 if ( !ok || factor < 2 ) continue;
112 mOverviews.append( { factor, parts[1] } );
113 }
114
115 // Sort by factor ascending so selectOverviewMap() scans coarsest-first
116 std::sort( mOverviews.begin(), mOverviews.end(),
117 []( const GrassRasterOverview &a, const GrassRasterOverview &b ){
118 return a.factor < b.factor; } );
119
120 if ( !mOverviews.isEmpty() )
121 qCInfo(lcCanvas) << "GrassRasterMapLayer:" << mName
122 << "— loaded" << mOverviews.size() << "overview levels";
123}

125// ---------------------------------------------------------------------------
126// selectOverviewMap — GDAL-style level selection.
127// Use the finest pre-built overview whose resolution is coarser than
128// or equal to the required output resolution.
129// Returns the native map name when no overview qualifies.
130// ---------------------------------------------------------------------------
131QString GrassRasterMapLayer::selectOverviewMap( double reqEwRes ) const
132{
133 if ( mOverviews.isEmpty() || mEwRes <= 0 ) return mName;
134
135 // Decimation factor the viewport is requesting
136 const int decimation = static_cast<int>( reqEwRes / mEwRes );
137 if ( decimation < 2 ) return mName;
138
139 // Largest overview factor that does not exceed the decimation
140 // (avoids upsampling artefacts)
141 QString best = mName;
142 for ( const GrassRasterOverview &ovr : mOverviews )
143 {
144 if ( ovr.factor <= decimation )
145 best = ovr.mapName; // keep the coarsest qualifying level
146 else
147 break; // sorted ascending — stop when we overshoot
148 }
149 return best;
150}

152// ---------------------------------------------------------------------------
153// render — Phase B: GRASS viewport-region resampling + overview selection.
154//
155// Complexity: O(outW × outH) regardless of native raster size.
156// With pre-built overview levels the number of native cells GRASS reads
157// from disk is proportional to the overview map (e.g. L8 = 64× smaller).
158// ---------------------------------------------------------------------------
159QImage GrassRasterMapLayer::render( int viewW, int viewH,
160 const GrassExtent &vp ) const
161{
162 if ( cacheHit( viewW, viewH, vp ) )
163 return cachedImage();
164
165 if ( !load() || viewW <= 0 || viewH <= 0 || !vp.isValid() )
166 return {};
167
168 // Lazy-load overview info
169 loadOverviewInfo();
170
171 // Intersection of viewport with raster extent
172 const double intN = std::min( vp.north, mExtent.north );
173 const double intS = std::max( vp.south, mExtent.south );
174 const double intE = std::min( vp.east, mExtent.east );
175 const double intW = std::max( vp.west, mExtent.west );
176
177 QImage img( viewW, viewH, QImage::Format_ARGB32_Premultiplied );
178 img.fill( Qt::transparent );
179
180 if ( intN <= intS || intE <= intW )
181 {
182 storeCache( viewW, viewH, vp, img );
183 return img;
184 }
185
186 // Destination rectangle within the output image
187 const int dstX = static_cast<int>( std::round( ( intW - vp.west ) / vp.width() * viewW ) );
188 const int dstY = static_cast<int>( std::round( ( vp.north - intN ) / vp.height() * viewH ) );
189 const int dstW = static_cast<int>( std::round( ( intE - intW ) / vp.width() * viewW ) );
190 const int dstH = static_cast<int>( std::round( ( intN - intS ) / vp.height() * viewH ) );
191
192 if ( dstW <= 0 || dstH <= 0 )
193 {
194 storeCache( viewW, viewH, vp, img );
195 return img;
196 }
197
198 // Select the best-fit overview map (may be the native map)
199 const double reqEwRes = ( intE - intW ) / dstW;
200 const QString renderMap = selectOverviewMap( reqEwRes );
201
202 const QByteArray nameBytes = renderMap.toLocal8Bit();
203 const QByteArray mapsetBytes = mMapset.toLocal8Bit();
204 const char *mapset = mapsetBytes.isEmpty() ? "" : mapsetBytes.constData();
205
206 // ── Rendering — Phase B ────────────────────────────────────────────────
207 // Set GRASS region = viewport intersection at output resolution.
208 // GRASS resamples the raster to (dstW × dstH) → O(outW × outH) reads.
209 //
210 // Near-global rasters (native EW extent ≥ 360°, e.g. bathy or the
211 // grid_float_All_2023_* series with east=180.003°) require a special
212 // safeE clamp. GRASS internally computes:
213 // native_cols = round( region_extent / native_ew_res )
214 // If region_extent is 360.0 exactly, floating-point gives
215 // round(360.0 / ew_res) = file_cols ± 1 (rounding direction varies).
216 // An off-by-one produces a mismatched row buffer → heap corruption → SIGABRT.
217 //
218 // Fix: clamp safeE so that region_extent = 360° - one_native_cell.
219 // native_cols = round((360 - ew_res) / ew_res) = round(file_cols - 1)
220 // = file_cols - 1 (safe: reads one fewer column, misses
221 // only the pixel at longitude > 180°).
222 // ── Phase B: viewport-resolution resampling ────────────────────────────
223 // GRASS resamples the raster to exactly (renderH × renderW) output cells.
224 //
225 // safeE clamping for near-global rasters (native EW extent ≥ 360°):
226 // GRASS internally computes native_cols = round(region_extent / ew_res).
227 // Setting region_extent = 360° causes floating-point rounding to give
228 // native_cols = file_cols ± 1. A +1 overcount → buffer overflow → SIGABRT.
229 // Fix: cap region_extent at (360° - one_native_cell) so the rounded value
230 // is always file_cols - 1 (safe undercount: we lose one pixel at lon > 180°).
231 struct Cell_head savedRegion;
232 G_get_window( &savedRegion );
233
234 // safeE clamping — ONLY for geographic (lat/lon) CRS.
235 // In PROJECTION_LL, extents are in degrees (0–360). Setting region_extent
236 // = 360° makes GRASS compute native_cols = file_cols ± 1 due to floating-
237 // point rounding; a +1 overcount → buffer overflow → SIGABRT.
238 // Fix: cap to 360° - one_native_cell.
239 //
240 // For projected CRS (UTM, State Plane, …) extents are in metres.
241 // 15 000 m > 359 → the old guard fired incorrectly, clamping the EW
242 // extent to 350 m and producing a badly stretched stripe pattern.
243 const double nativeEwExtent = mExtent.east - mExtent.west;
244 double safeE = intE;
245 if ( savedRegion.proj == PROJECTION_LL )
246 {
247 if ( nativeEwExtent > 359.0 && mEwRes > 0 )
248 {
249 const double maxSafeExt = 360.0 - mEwRes;
250 if ( intE - intW > maxSafeExt )
251 safeE = intW + maxSafeExt;
252 }
253 else if ( intE - intW > 360.0 )
254 {
255 safeE = intW + 360.0;
256 }
257 }
258
259 // ── Build the render window ───────────────────────────────────────────
260 // Set the window to viewport resolution — GRASS handles resampling via
261 // nearest-neighbour internally (same as wxPython d.rast). One GRASS cell
262 // per output pixel. Simple, correct, and matches reference rendering.
263 const double vpNsRes = ( intN - intS ) / dstH;
264 const double vpEwRes = ( safeE - intW ) / dstW;
265
266 struct Cell_head vpRegion = savedRegion;
267 vpRegion.north = intN;
268 vpRegion.south = intS;
269 vpRegion.east = safeE;
270 vpRegion.west = intW;
271 vpRegion.ns_res = vpNsRes;
272 vpRegion.ew_res = vpEwRes;
273 vpRegion.rows = 0;
274 vpRegion.cols = 0;
275
276 // Safety clamp for geographic locations.
277 if ( savedRegion.proj == PROJECTION_LL )
278 {
279 vpRegion.north = std::min( vpRegion.north, 90.0 );
280 vpRegion.south = std::max( vpRegion.south, -90.0 );
281 if ( vpRegion.north <= vpRegion.south || vpRegion.east <= vpRegion.west )
282 {
283 Rast_set_window( &savedRegion );
284 storeCache( viewW, viewH, vp, img );
285 return img;
286 }
287 vpRegion.ns_res = ( vpRegion.north - vpRegion.south ) / dstH;
288 }
289
290 Rast_set_window( &vpRegion );
291 const int renderH = vpRegion.rows; // output rows (≈ dstH)
292 const int renderW = vpRegion.cols; // output cols (≈ dstW)
293
294 struct Colors colors;
295 if ( Rast_read_colors( nameBytes.constData(), mapset, &colors ) < 0 )
296 {
297 // No color table — build a grey scale from the raster's actual data range.
298 RASTER_MAP_TYPE rtype = Rast_map_type( nameBytes.constData(), mapset );
299 CELL cmin = 0, cmax = 255;
300 if ( rtype == CELL_TYPE )
301 {
302 struct Range range;
303 Rast_init_range( &range );
304 if ( Rast_read_range( nameBytes.constData(), mapset, &range ) >= 0 )
305 Rast_get_range_min_max( &range, &cmin, &cmax );
306 }
307 else
308 {
309 struct FPRange fprange;
310 Rast_init_fp_range( &fprange );
311 if ( Rast_read_fp_range( nameBytes.constData(), mapset, &fprange ) >= 0 )
312 {
313 DCELL dmin, dmax;
314 Rast_get_fp_range_min_max( &fprange, &dmin, &dmax );
315 cmin = static_cast<CELL>( dmin );
316 cmax = static_cast<CELL>( dmax );
317 }
318 }
319 if ( cmax <= cmin ) cmax = cmin + 255;
320 Rast_make_grey_scale_colors( &colors, cmin, cmax );
321 }
322
323 const int fd = Rast_open_old( nameBytes.constData(), mapset );
324 if ( fd < 0 )
325 {
326 mError = QString( "Cannot open raster '%1'" ).arg( renderMap );
327 Rast_free_colors( &colors );
328 Rast_set_window( &savedRegion );
329 storeCache( viewW, viewH, vp, img );
330 return img;
331 }
332
333 auto *red = new unsigned char[ renderW ];
334 auto *green = new unsigned char[ renderW ];
335 auto *blue = new unsigned char[ renderW ];
336 auto *nul = new unsigned char[ renderW ];
337
338 // One GRASS row per output row — GRASS handles resampling internally.
339 for ( int row = 0; row < renderH; ++row )
340 {
341 const int imgRow = dstY + row;
342 if ( imgRow < 0 || imgRow >= viewH ) continue;
343 Rast_get_row_colors( fd, row, &colors, red, green, blue, nul );
344 QRgb *scanline = reinterpret_cast<QRgb *>( img.scanLine( imgRow ) );
345 for ( int col = 0; col < renderW; ++col )
346 {
347 const int imgCol = dstX + col;
348 if ( imgCol < 0 || imgCol >= viewW ) continue;
349 if ( !nul[col] )
350 scanline[imgCol] = qRgba( red[col], green[col], blue[col], 255 );
351 }
352 }
353
354 Rast_close( fd );
355 Rast_free_colors( &colors );
356 delete[] red; delete[] green; delete[] blue; delete[] nul;
357 Rast_set_window( &savedRegion );
358
359 storeCache( viewW, viewH, vp, img );
360 return img;
