54QByteArray GrassXmlParser::fetchXml( const QString &moduleName,
55 QString *errorOut )
56{
57 const QString bin = findModuleBin( moduleName );
58 if ( bin.isEmpty() )
59 {
60 if ( errorOut )
61 *errorOut = QString( "Module '%1' not found in GISBASE/bin or GISBASE/scripts." )
62 .arg( moduleName );
63 return {};
64 }
65
66 // Build environment with GISRC file so the subprocess can call G_gisinit().
67 // qtgrass uses G_GISRC_MODE_MEMORY; child processes need a real GISRC file.
68 QProcessEnvironment env = GrassEnv::instance()->buildSubprocessEnv();
69
70 QProcess proc;
71 proc.setProcessEnvironment( env );
72 proc.start( bin, { "--interface-description" } );
73
74 if ( !proc.waitForStarted( 5000 ) )
75 {
76 if ( errorOut )
77 *errorOut = "Failed to start: " + proc.errorString();
78 return {};
79 }
80 if ( !proc.waitForFinished( 30000 ) )
81 {
82 proc.kill();
83 if ( errorOut )
84 *errorOut = "Timed out waiting for --interface-description";
85 return {};
86 }
87
88 const QByteArray out = proc.readAllStandardOutput();
89 if ( out.isEmpty() )
90 {
91 if ( errorOut )
92 *errorOut = "No output from --interface-description: "
93 + proc.readAllStandardError();
94 }
95 return out;
96}

101GrassModuleDesc GrassXmlParser::parse( const QByteArray &xml )
102{
103 GrassModuleDesc desc;
104 if ( xml.isEmpty() )
105 return desc;
106
107 QDomDocument doc;
108 auto parseResult = doc.setContent( xml );
109 if ( !parseResult )
110 {
111 qCWarning(lcModule) << "GrassXmlParser: XML parse error:" << parseResult.errorMessage;
112 return desc;
113 }
114
115 const QDomElement task = doc.documentElement(); // <task name="...">
116 if ( task.isNull() )
117 return desc;
118
119 desc.name = task.attribute( "name" );
120 desc.label = elemText( task, "label" );
121 desc.description = elemText( task, "description" );
122 desc.keywords = elemText( task, "keywords" );
123
124 // --- Parameters ---
125 QDomNodeList paramNodes = task.elementsByTagName( "parameter" );
126 for ( int i = 0; i < paramNodes.count(); ++i )
127 {
128 QDomElement pe = paramNodes.at( i ).toElement();
129 GrassParamDesc p;
130 p.name = pe.attribute( "name" );
131 p.type = pe.attribute( "type", "string" );
132 p.required = ( pe.attribute( "required" ) == "yes" );
133 p.multiple = ( pe.attribute( "multiple" ) == "yes" );
134
135 p.label = elemText( pe, "label" );
136 p.description = elemText( pe, "description" );
137 p.guiSection = elemText( pe, "guisection" );
138 p.defaultValue= elemText( pe, "default" );
139
140 // gisprompt
141 QDomElement gp = pe.firstChildElement( "gisprompt" );
142 if ( !gp.isNull() )
143 {
144 p.hasGisprompt = true;
145 p.gisprompt.age = gp.attribute( "age" );
146 p.gisprompt.element = gp.attribute( "element" );
147 p.gisprompt.prompt = gp.attribute( "prompt" );
148 }
149
150 // enum values
151 QDomElement valuesElem = pe.firstChildElement( "values" );
152 if ( !valuesElem.isNull() )
153 {
154 QDomNodeList vals = valuesElem.elementsByTagName( "value" );
155 for ( int j = 0; j < vals.count(); ++j )
156 {
157 QDomElement ve = vals.at( j ).toElement();
158 const QString vname = elemText( ve, "name" );
159 const QString vdesc = elemText( ve, "description" );
160 p.values.append( { vname, vdesc } );
161 }
162 }
163
164 desc.params.append( p );
165 }
166
167 // --- Flags ---
168 QDomNodeList flagNodes = task.elementsByTagName( "flag" );
169 for ( int i = 0; i < flagNodes.count(); ++i )
170 {
171 QDomElement fe = flagNodes.at( i ).toElement();
172 GrassFlagDesc f;
173 f.name = fe.attribute( "name" );
174 f.label = elemText( fe, "label" );
175 f.description = elemText( fe, "description" );
176 f.guiSection = elemText( fe, "guisection" );
177 f.hidden = isInternalFlag( f.name );
178 desc.flags.append( f );
179 }
180
181 return desc;
182}
