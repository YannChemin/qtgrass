279inline const OsmCartoPreset* findOsmPreset( const QString &key )
280{
281 for ( const auto &p : osmCartoLines() ) if ( p.key == key ) return &p;
282 for ( const auto &p : osmCartoAreas() ) if ( p.key == key ) return &p;
283 for ( const auto &p : osmCartoPoints() ) if ( p.key == key ) return &p;
284 return nullptr;
285}
