1#pragma once
2// GrassAddonMenu::populate()
3//
4// Reads the compile-time list generated from ADDONS_MENU.md (the human-edited
5// authoritative placement table) and inserts each addon into the correct QMenu.
6//
7// Menu path format (from ADDONS_MENU.md second column):
8// "imagery/evapotranspiration" → Imagery → Evapotranspiration (existing submenu)
9// "raster/multi-criteria-decision" → Raster → new submenu created if not yet present
10// "tools/addons" → Tools → Addon Extensions
11//
12// Submenus that match an existing QMenu (via PrefixMenuMap) are used directly.
13// Submenus with no existing match are created at the bottom of the parent menu.
14//
15// Each action is labelled:
16// "Human readable description (module.name)"
17// or just the module name when no description was found.

39// Slug → human-readable submenu title used when creating a NEW submenu.
40// E.g. "multi-criteria-decision" → "Multi-criteria decision"
41static inline QString slugToTitle( const QString &slug )
42{
43 QString s = slug;
44 s.replace('-', ' ');
45 if ( !s.isEmpty() ) s[0] = s[0].toUpper();
46 return s;
47}

50static inline void addAction( QMenu *menu, const QString &name,

57 QAction *a = menu->addAction( text );
58 a->setToolTip( name );

60 QObject::connect( a, &QAction::triggered, [cb, m] { cb(m); } );

65static inline void populate(
66 const GrassMenuBuilder::PrefixMenuMap &prefixMap,
67 const GrassMenuBuilder::CommandActionMap &cmdMap,
68 QMenu *toolsMenu,
69 const ModuleCallback &callback )
70{
71#ifndef HAVE_GRASS_ADDONS
72 Q_UNUSED(prefixMap) Q_UNUSED(toolsMenu) Q_UNUSED(callback)
73 return;
74#else
75 // ── Pass 0: "after:base_module" — numbered variants ──────────────
76 // Insert each addon with menuPath "after:X" immediately below the
77 // core module X's QAction, in the same QMenu that hosts X.
78 for ( const GrassAddons::AddonInfo &info : GrassAddons::allAddons() )
79 {
80 const QString fullPath = info.menuPath;
81 if ( !fullPath.startsWith( "after:", Qt::CaseInsensitive ) )
82 continue;
83
84 const QString baseCmd = fullPath.mid( 6 ).trimmed(); // "after:i.atcorr" → "i.atcorr"
85 auto loc = cmdMap.value( baseCmd );
86 if ( !loc.menu || !loc.action ) continue; // base module not in menu
87
88 QAction *newAct = new QAction(
89 QString( "%1 (%2)" ).arg( info.label[0] ? info.label : info.name )
90 .arg( info.name ),
91 loc.menu );
92 newAct->setToolTip( info.name );
93 const QString m = info.name;
94 QObject::connect( newAct, &QAction::triggered, [callback, m] { callback( m ); } );
95
96 // Insert immediately after the base module's action
97 const QList<QAction*> acts = loc.menu->actions();
98 const int idx = acts.indexOf( loc.action );
99 if ( idx >= 0 && idx + 1 < acts.size() )
100 loc.menu->insertAction( acts[ idx + 1 ], newAct );
101 else
102 loc.menu->addAction( newAct );
103 }
104
105 // ── Build the top-level menu map from prefixMap ───────────────────
106 // prefixMap contains "i" → Imagery menu, "r" → Raster menu, etc.
107 // Build a separate slug→QMenu map for top-level lookups:
108 // "imagery" → Imagery QMenu, "raster" → Raster, …
109 static const QMap<QString,QString> catToSlug = {
110 { "r", "raster" }, { "r3", "raster-3d" }, { "v", "vector" },
111 { "i", "imagery" }, { "t", "temporal" }, { "db", "database" },
112 { "d", "settings" }, { "g", "settings" }, { "m", "tools" },
113 };
114 QMap<QString, QMenu*> topMenus; // slug → QMenu*
115 for ( auto it = catToSlug.cbegin(); it != catToSlug.cend(); ++it )
116 {
117 QMenu *m = prefixMap.value( it.key() );
118 if ( m && !topMenus.contains( it.value() ) )
119 topMenus.insert( it.value(), m );
120 }
121 if ( toolsMenu ) topMenus.insert( "tools", toolsMenu );
122
123 // ── Per-top-menu: track created submenus & separator state ────────
124 // submenus created this session: slug → QMenu*
125 QMap<QString, QMenu*> createdSubs;
126
127 // Track which menus already got a separator+header for addon additions
128 // to EXISTING submenus (ones found via prefixMap two-level match).
129 QSet<QMenu*> headerAdded;
130
131 // ── Process each addon from the generated header ──────────────────
132 for ( const GrassAddons::AddonInfo &info : GrassAddons::allAddons() )
133 {
134 const QString name = info.name;
135 const QString label = info.label;
136 const QString fullPath = info.menuPath; // e.g. "imagery/evapotranspiration"
137
138 // Already handled in Pass 0
139 if ( fullPath.startsWith( "after:", Qt::CaseInsensitive ) ) continue;
140
141 // Split into top-level slug and optional submenu slug
142 const QStringList parts = fullPath.split('/');
143 if ( parts.isEmpty() ) continue;
144
145 const QString topSlug = parts[0]; // "imagery"
146 const QString subSlug = parts.size() > 1 ? parts[1] : ""; // "evapotranspiration"
147
148 QMenu *topMenu = topMenus.value( topSlug, toolsMenu );
149 if ( !topMenu ) topMenu = toolsMenu;
150 if ( !topMenu ) continue;
151
152 if ( subSlug.isEmpty() )
153 {
154 // No submenu — add directly to top-level menu
155 addAction( topMenu, name, label, callback );
156 continue;
157 }
158
159 // ── Try to find an existing submenu via prefixMap ─────────────
160 // The first token of the module name gives us the two-level prefix:
161 // "i.evapo.potrad" → "i.evapo". If that's in prefixMap, use it.
162 QMenu *targetSub = nullptr;
163
164 const QStringList modParts = name.split('.');
165 if ( modParts.size() >= 2 )
166 {
167 const QString twoLevel = modParts[0] + "." + modParts[1];
168 targetSub = prefixMap.value( twoLevel, nullptr );
169 }
170
171 if ( targetSub )
172 {
173 // Existing submenu — add with "— Addon modules —" separator once
174 if ( !headerAdded.contains( targetSub ) )
175 {
176 targetSub->addSeparator();
177 QAction *hdr = targetSub->addAction( "— Addon modules —" );
178 hdr->setEnabled( false );
179 headerAdded.insert( targetSub );
180 }
181 addAction( targetSub, name, label, callback );
182 }
183 else
184 {
185 // No existing submenu — find or create one from the slug
186 const QString subKey = topSlug + "/" + subSlug;
187 if ( !createdSubs.contains( subKey ) )
188 {
189 // Check whether the top menu already has a submenu with this
190 // title (added in a previous loop iteration)
191 const QString title = slugToTitle( subSlug );
192 QMenu *existing = nullptr;
193 for ( QAction *a : topMenu->actions() )
194 if ( a->menu() && a->menu()->title() == title )
195 { existing = a->menu(); break; }
196
197 if ( existing )
198 {
199 createdSubs.insert( subKey, existing );
200 }
201 else
202 {
203 // Create the new submenu, preceded by a separator if this
204 // is the first new submenu in this top-level menu.
205 if ( !createdSubs.keys().filter( topSlug + "/" ).isEmpty()
206 || !headerAdded.contains( topMenu ) )
207 {
208 // Add separator before the very first new submenu block
209 if ( !headerAdded.contains( topMenu ) )
210 {
211 topMenu->addSeparator();
212 headerAdded.insert( topMenu );
213 }
214 }
215 QMenu *sub = topMenu->addMenu( title );
216 createdSubs.insert( subKey, sub );
217 }
218 }
219
220 addAction( createdSubs[subKey], name, label, callback );
221 }
222 }
223#endif
224}
