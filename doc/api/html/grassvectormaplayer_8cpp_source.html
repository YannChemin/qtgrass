17// ── Crash guard for Vect_open_old ──────────────────────────────────────────
18// Corrupt topology binary data can cause fread() to dereference a bad pointer
19// deep inside dig_load_plus → dig_Rd_P_isle. We wrap Vect_open_old in a
20// sigsetjmp region so the process survives bad topo files and just skips the
21// map with a warning (user should run: v.build <map>).
22static sigjmp_buf sVectJmpBuf;
23static volatile sig_atomic_t sVectJmpActive = 0;
24
25static void vectCrashGuard( int )
26{
27 if ( sVectJmpActive )
28 {
29 sVectJmpActive = 0;
30 siglongjmp( sVectJmpBuf, 1 );
31 }
32 // Outside protected section — restore default and re-raise so we still
33 // get a core dump for any unrelated crash.
34 struct sigaction sa {};
35 sa.sa_handler = SIG_DFL;
36 sigaction( SIGSEGV, &sa, nullptr );
37 raise( SIGSEGV );
38}

49bool GrassVectorMapLayer::load()
50{
51 mLoaded = true;
52
53 const QByteArray nameBytes = mName.toLocal8Bit();
54 const QByteArray mapsetBytes = mMapset.toLocal8Bit();
55 const char *ms = mapsetBytes.isEmpty() ? "" : mapsetBytes.constData();
56
57 // Pre-check: G_find_vector2() returns NULL without calling G_fatal_error
58 // when the map is absent, so we can handle missing maps gracefully before
59 // Vect_open_old() has a chance to abort the process.
60 QByteArray mutableName = nameBytes;
61 if ( !G_find_vector2( mutableName.data(), ms ) )
62 {
63 mError = QString( "Vector map '%1' not found in mapset '%2'" )
64 .arg( mName, ms[0] ? ms : "(current)" );
65 qCWarning(lcVector) << "GrassVectorMapLayer:" << mError;
66 return false;
67 }
68
69 // Vect_open_old can SIGSEGV when the topo binary is internally corrupt
70 // (e.g. dig_Rd_P_isle reads a bad pointer from corrupted isle data).
71 // Catch it with sigsetjmp so we can skip the map and warn the user.
72 struct Map_info map;
73 {
74 struct sigaction newSa {}, oldSa {};
75 newSa.sa_handler = vectCrashGuard;
76 sigemptyset( &newSa.sa_mask );
77 newSa.sa_flags = 0;
78 sigaction( SIGSEGV, &newSa, &oldSa );
79 sVectJmpActive = 1;
80
81 bool openOk = false;
82 if ( sigsetjmp( sVectJmpBuf, 1 ) == 0 )
83 {
84 openOk = ( Vect_open_old( &map, nameBytes.constData(), ms ) >= 0 );
85 }
86 else
87 {
88 // SIGSEGV inside Vect_open_old — topo file is corrupt
89 mError = QString( "Corrupt topology for '%1' — run: v.build map=%1" )
90 .arg( mName );
91 qCWarning(lcVector) << "GrassVectorMapLayer:" << mError;
92 }
93
94 sVectJmpActive = 0;
95 sigaction( SIGSEGV, &oldSa, nullptr );
96
97 if ( !openOk )
98 {
99 if ( mError.isEmpty() )
100 {
101 mError = QString( "Cannot open vector '%1'" ).arg( mName );
102 qCWarning(lcVector) << "GrassVectorMapLayer:" << mError;
103 }
104 return false;
105 }
106 }
107
108 // Bounding box
109 struct bound_box bbox;
110 Vect_get_map_box( &map, &bbox );
111 mExtent = { bbox.N, bbox.S, bbox.E, bbox.W };
112
113 struct line_pnts *pnts = Vect_new_line_struct();
114 struct line_cats *cats = Vect_new_cats_struct();
115
116 const int nlines = static_cast<int>( Vect_get_num_lines( &map ) );
117
118 for ( int i = 1; i <= nlines; ++i )
119 {
120 const int type = Vect_read_line( &map, pnts, cats, i );
121 if ( type < 0 ) continue;
122
123 if ( type & GV_POINTS ) // GV_POINT | GV_CENTROID
124 {
125 for ( int k = 0; k < pnts->n_points; ++k )
126 mPoints.append( { pnts->x[k], pnts->y[k] } );
127 }
128 else if ( type & GV_LINES ) // GV_LINE | GV_BOUNDARY
129 {
130 QPolygonF poly;
131 poly.reserve( pnts->n_points );
132 for ( int k = 0; k < pnts->n_points; ++k )
133 poly.append( { pnts->x[k], pnts->y[k] } );
134 mLines.append( poly );
135 }
136 }
137
138 // Areas — iterate via area IDs
139 const int nareas = static_cast<int>( Vect_get_num_areas( &map ) );
140 for ( int a = 1; a <= nareas; ++a )
141 {
142 if ( Vect_get_area_points( &map, a, pnts ) < 0 ) continue;
143 QPolygonF poly;
144 poly.reserve( pnts->n_points );
145 for ( int k = 0; k < pnts->n_points; ++k )
146 poly.append( { pnts->x[k], pnts->y[k] } );
147 if ( !poly.isEmpty() )
148 mAreas.append( poly );
149 }
150
151 Vect_destroy_line_struct( pnts );
152 Vect_destroy_cats_struct( cats );
153 Vect_close( &map );
154
155 qCInfo(lcVector) << "GrassVectorMapLayer:" << mName
156 << "lines:" << mLines.size()
157 << "areas:" << mAreas.size()
158 << "points:" << mPoints.size();
159 return true;
160}

165QImage GrassVectorMapLayer::render( int viewW, int viewH,
166 const GrassExtent &vp ) const
167{
168 if ( cacheHit( viewW, viewH, vp ) )
169 return cachedImage();
170
171 if ( !mLoaded )
172 const_cast<GrassVectorMapLayer *>( this )->load();
173
174 if ( mLines.isEmpty() && mAreas.isEmpty() && mPoints.isEmpty() )
175 return {};
176
177 if ( viewW <= 0 || viewH <= 0 || !vp.isValid() ) return {};
178
179 // Helper: geographic (east, north) → pixel
180 auto geo2px = [&]( double e, double n ) -> QPointF {
181 return {
182 ( e - vp.west ) / vp.width() * viewW,
183 ( vp.north - n ) / vp.height() * viewH
184 };
185 };
186
187 auto projectPoly = [&]( const QPolygonF &geo ) -> QPolygonF {
188 QPolygonF px;
189 px.reserve( geo.size() );
190 for ( const QPointF &p : geo )
191 px.append( geo2px( p.x(), p.y() ) );
192 return px;
193 };
194
195 QImage img( viewW, viewH, QImage::Format_ARGB32_Premultiplied );
196 img.fill( Qt::transparent );
197
198 QPainter p( &img );
199 p.setRenderHint( QPainter::Antialiasing );
200
201 // ── Areas (filled polygons) ───────────────────────────────────────────────
202 // drawFill → fill interior; drawLines → draw border
203 if ( !mAreas.isEmpty() && ( mStyle.drawFill || mStyle.drawLines ) )
204 {
205 const QPen border = mStyle.drawLines
206 ? QPen( mStyle.lineColor, mStyle.lineWidth, mStyle.penStyle )
207 : Qt::NoPen;
208 const QBrush fill = mStyle.drawFill
209 ? QBrush( mStyle.fillColor, mStyle.brushStyle )
210 : Qt::NoBrush;
211 p.setPen( border );
212 p.setBrush( fill );
213 for ( const QPolygonF &geo : mAreas )
214 p.drawPolygon( projectPoly( geo ) );
215 }
216
217 // ── Lines ────────────────────────────────────────────────────────────────
218 if ( mStyle.drawLines && !mLines.isEmpty() )
219 {
220 QPen pen( mStyle.lineColor, mStyle.lineWidth, mStyle.penStyle );
221 pen.setCapStyle( Qt::RoundCap );
222 pen.setJoinStyle( Qt::RoundJoin );
223 p.setPen( pen );
224 p.setBrush( Qt::NoBrush );
225 for ( const QPolygonF &geo : mLines )
226 p.drawPolyline( projectPoly( geo ) );
227 }
228
229 // ── Points ───────────────────────────────────────────────────────────────
230 if ( mStyle.drawPoints && !mPoints.isEmpty() )
231 {
232 const QColor &pc = mStyle.pointColor;
233 const float r = mStyle.pointSize / 2.0f;
234
235 auto drawShape = [&]( const QPointF &px )
236 {
237 p.save();
238 p.translate( px );
239
240 switch ( mStyle.pointShape )
241 {
242 case OsmPointShape::Circle:
243 p.setPen( Qt::NoPen );
244 p.setBrush( pc );
245 p.drawEllipse( QPointF(0,0), r, r );
246 break;
247
248 case OsmPointShape::Square:
249 p.setPen( Qt::NoPen );
250 p.setBrush( pc );
251 p.drawRect( QRectF( -r, -r, 2*r, 2*r ) );
252 break;
253
254 case OsmPointShape::Triangle:
255 {
256 const QPointF pts[3] = { {0,-r}, {-r,r}, {r,r} };
257 p.setPen( Qt::NoPen );
258 p.setBrush( pc );
259 p.drawPolygon( pts, 3 );
260 break;
261 }
262 case OsmPointShape::Diamond:
263 {
264 const QPointF pts[4] = { {0,-r}, {r,0}, {0,r}, {-r,0} };
265 p.setPen( Qt::NoPen );
266 p.setBrush( pc );
267 p.drawPolygon( pts, 4 );
268 break;
269 }
270 case OsmPointShape::Cross:
271 {
272 const float w = r * 0.35f;
273 p.setPen( QPen( pc, w*2, Qt::SolidLine, Qt::RoundCap ) );
274 p.drawLine( QPointF(-r,0), QPointF(r,0) );
275 p.drawLine( QPointF(0,-r), QPointF(0,r) );
276 break;
277 }
278 case OsmPointShape::Star:
279 {
280 const float ri = r * 0.45f;
281 QPolygonF star;
282 for ( int i = 0; i < 10; ++i )
283 {
284 const float angle = i * M_PI / 5 - M_PI/2;
285 const float rad = ( i%2==0 ) ? r : ri;
286 star << QPointF( rad*std::cos(angle), rad*std::sin(angle) );
287 }
288 p.setPen( Qt::NoPen );
289 p.setBrush( pc );
290 p.drawPolygon( star );
291 break;
292 }
293 case OsmPointShape::MedicalCross:
294 {
295 const float arm = r * 0.4f;
296 // White background circle
297 p.setPen( Qt::NoPen );
298 p.setBrush( Qt::white );
299 p.drawEllipse( QPointF(0,0), r, r );
300 // Coloured cross arms
301 p.setBrush( pc );
302 p.drawRect( QRectF(-arm, -r, 2*arm, 2*r) );
303 p.drawRect( QRectF(-r, -arm, 2*r, 2*arm) );
304 break;
305 }
306 case OsmPointShape::ArrowUp:
307 {
308 const QPointF pts[3] = { {0,-r}, {-r,r*0.5f}, {r,r*0.5f} };
309 p.setPen( Qt::NoPen );
310 p.setBrush( pc );
311 p.drawPolygon( pts, 3 );
312 p.fillRect( QRectF(-r*0.25f, r*0.2f, r*0.5f, r*0.6f), pc );
313 break;
314 }
315 case OsmPointShape::Pentagon:
316 {
317 QPolygonF pent;
318 for ( int i = 0; i < 5; ++i )
319 {
320 const float angle = i * 2*M_PI/5 - M_PI/2;
321 pent << QPointF( r*std::cos(angle), r*std::sin(angle) );
322 }
323 p.setPen( Qt::NoPen );
324 p.setBrush( pc );
325 p.drawPolygon( pent );
326 break;
327 }
328 case OsmPointShape::Hexagon:
329 {
330 QPolygonF hex;
331 for ( int i = 0; i < 6; ++i )
332 {
333 const float angle = i * M_PI/3;
334 hex << QPointF( r*std::cos(angle), r*std::sin(angle) );
335 }
336 p.setPen( Qt::NoPen );
337 p.setBrush( pc );
338 p.drawPolygon( hex );
339 break;
340 }
341 }
342
343 // Thin contrasting outline on all solid shapes
344 if ( mStyle.pointShape != OsmPointShape::Cross &&
345 mStyle.pointShape != OsmPointShape::MedicalCross )
346 {
347 p.setBrush( Qt::NoBrush );
348 p.setPen( QPen( pc.darker(160), 0.8f ) );
349 switch ( mStyle.pointShape )
350 {
351 case OsmPointShape::Circle:
352 p.drawEllipse( QPointF(0,0), r, r ); break;
353 case OsmPointShape::Square:
354 p.drawRect( QRectF(-r,-r,2*r,2*r) ); break;
355 default: break;
356 }
357 }
358
359 p.restore();
360 };
361
362 for ( const QPointF &gpt : mPoints )
363 drawShape( geo2px( gpt.x(), gpt.y() ) );
364 }
365
366 p.end();
367 storeCache( viewW, viewH, vp, img );
368 return img;
