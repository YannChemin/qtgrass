91void GrassWmsBrowserDialog::fetchCapabilities()
92{
93 QString url = mUrlEdit->text().trimmed();
94 if (url.isEmpty()) return;
95
96 // Strip any existing query and add GetCapabilities
97 const int q = url.indexOf('?');
98 const QString base = (q >= 0) ? url.left(q) : url;
99 const QString capUrl = base +
100 "?SERVICE=WMS&VERSION=1.3.0&REQUEST=GetCapabilities";
101
102 mStatusLabel->setText("Fetching capabilities from:\n" + capUrl);
103 mLayerList->clear();
104 mLayers.clear();
105
106 QUrl qurl{capUrl};
107 QNetworkRequest req{qurl};
108 req.setAttribute(QNetworkRequest::RedirectPolicyAttribute,
109 QNetworkRequest::NoLessSafeRedirectPolicy);
110 QNetworkReply *reply = mNam->get(req);
111
112 connect(reply, &QNetworkReply::finished, this, [this, reply] {
113 reply->deleteLater();
114 if (reply->error() != QNetworkReply::NoError)
115 {
116 mStatusLabel->setText("Error: " + reply->errorString());
117 return;
118 }
119 parseCapabilities(reply->readAll());
120 });
121}

148void GrassWmsBrowserDialog::parseLayers(const QDomElement &parent,
149 QVector<WmsLayer> &out)
150{
151 for (QDomElement el = parent.firstChildElement("Layer");
152 !el.isNull(); el = el.nextSiblingElement("Layer"))
153 {
154 WmsLayer l;
155 l.name = el.firstChildElement("Name").text().trimmed();
156 l.title = el.firstChildElement("Title").text().trimmed();
157
158 for (QDomElement crs = el.firstChildElement("CRS");
159 !crs.isNull(); crs = crs.nextSiblingElement("CRS"))
160 l.crsList.append(crs.text().trimmed());
161 for (QDomElement srs = el.firstChildElement("SRS");
162 !srs.isNull(); srs = srs.nextSiblingElement("SRS"))
163 l.crsList.append(srs.text().trimmed());
164 l.crsList.removeDuplicates();
165
166 if (!l.name.isEmpty()) out.append(l);
167 parseLayers(el, out); // recurse into nested Layer elements
168 }

233 if (p.exitCode() != 0)
234 mStatusLabel->setText("Error: " + out.left(300));
