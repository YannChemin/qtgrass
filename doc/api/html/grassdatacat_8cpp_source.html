71QStringList GrassDataCatalog::listMaps( const QString &type ) const
72{
73 const QString gisdb = QString::fromLocal8Bit( qgetenv( "GISDBASE" ) );
74 const QString loc = QString::fromLocal8Bit( qgetenv( "LOCATION_NAME" ) );
75 if ( gisdb.isEmpty() || loc.isEmpty() ) return {};
76
77 // Subdirectory within each mapset that holds this element type.
78 // For rasters we use cellhd/ (present for CELL, FCELL and DCELL alike).
79 QString subdir;
80 bool isDir = false;
81 if ( type == "raster" ) { subdir = "cellhd"; isDir = false; }
82 else if ( type == "vector" ) { subdir = "vector"; isDir = true; }
83 else if ( type == "raster_3d" ) { subdir = "grid3"; isDir = true; }
84 else if ( type == "region" ) { subdir = "windows"; isDir = false; }
85 else return {};
86
87 const QDir locationDir( gisdb + "/" + loc );
88 if ( !locationDir.exists() ) return {};
89
90 QStringList result;
91 const QStringList mapsets = locationDir.entryList(
92 QDir::Dirs | QDir::NoDotAndDotDot );
93
94 for ( const QString &mapset : mapsets )
95 {
96 const QDir mapsDir( gisdb + "/" + loc + "/" + mapset + "/" + subdir );
97 if ( !mapsDir.exists() ) continue;
98
99 const QDir::Filters f = isDir
100 ? ( QDir::Dirs | QDir::NoDotAndDotDot )
101 : ( QDir::Files | QDir::NoDotAndDotDot );
102
103 const QStringList entries = mapsDir.entryList( f );
104 for ( const QString &name : entries )
105 result << name + "@" + mapset;
106 }
107
108 result.sort( Qt::CaseInsensitive );
109 return result;
110}

115void GrassDataCatalog::populate( const QString &type, QTreeWidgetItem *parent )
116{
117 parent->takeChildren();
118 const QStringList maps = listMaps( type );
119 for ( const QString &name : maps )
120 {
121 const QString n = name.trimmed();
122 if ( n.isEmpty() ) continue;
123 auto *child = new QTreeWidgetItem( parent, { n } );
124 child->setData( 0, Qt::UserRole, type ); // "raster" or "vector"
125 child->setData( 0, Qt::UserRole + 1, n ); // full name (may include @mapset)
126 child->setToolTip( 0, type + ": " + n );
127 parent->addChild( child );
128 }
129 // Update count in group label
130 const QString baseLabel = parent->data( 0, Qt::UserRole + 2 ).toString();
131 parent->setText( 0, QString( "%1 (%2)" ).arg( baseLabel ).arg( maps.size() ) );
132}

160void GrassDataCatalog::onItemDoubleClicked( QTreeWidgetItem *item, int )
161{
162 const QString type = item->data( 0, Qt::UserRole ).toString();
163 const QString name = item->data( 0, Qt::UserRole + 1 ).toString();
164 if ( type.isEmpty() || name.isEmpty() ) return; // group node
165
166 // Parse "name@mapset" if present
167 const int at = name.indexOf( '@' );
168 const QString mapName = at < 0 ? name : name.left( at );
169 const QString mapset = at < 0 ? QString() : name.mid( at + 1 );
170
171 if ( type == "raster" )
172 mPanel->addRasterLayer( mapName, mapset );
173 else if ( type == "vector" )
174 mPanel->addVectorLayer( mapName, mapset );
175 // region / raster_3d: no canvas rendering yet
176}

181void GrassDataCatalog::onFilterChanged( const QString &text )
182{
183 const QString lower = text.toLower();
184 for ( int g = 0; g < mTree->topLevelItemCount(); ++g )
185 {
186 QTreeWidgetItem *group = mTree->topLevelItem( g );
187 int visible = 0;
188 for ( int c = 0; c < group->childCount(); ++c )
189 {
190 QTreeWidgetItem *child = group->child( c );
191 const bool match = lower.isEmpty() ||
192 child->text( 0 ).contains( lower, Qt::CaseInsensitive );
193 child->setHidden( !match );
194 if ( match ) ++visible;
195 }
196 group->setHidden( visible == 0 && !lower.isEmpty() );
197 }
