540// ---------------------------------------------------------------------------
541// buildRasterOverviews — right-click action for raster layers.
542//
543// Creates cascaded reduced-resolution copies of the raster entirely within
544// the GRASS storage structure (no GeoTIFF export):
545// L2 = r.resamp.stats input=MAP output=MAP_L2 (2× coarser)
546// L4 = r.resamp.stats input=MAP_L2 output=MAP_L4 (cascade, cheap)
547// L8 = r.resamp.stats input=MAP_L4 output=MAP_L8 (cascade, very cheap)
548// …
549// Total extra I/O ≈ 33% of the original raster (geometric series).
550//
551// Results are stored as native GRASS rasters in the same mapset.
552// A marker file cell_misc/<name>/qtgrass_ovr lists the available levels.
553// The renderer reads this marker and selects the finest qualifying level.
554// ---------------------------------------------------------------------------
555void GrassLayerTree::buildRasterOverviews( const QString &name,
556 const QString &mapset )
557{
558 // --- Gather session info ---
559 // The 'grass' startup script lives outside GISBASE (e.g. /usr/local/bin or
560 // /usr/bin) — find it via PATH, not by constructing from GISBASE.
561 const QString grassBin = QStandardPaths::findExecutable( "grass" );
562 const QString gisdb = QString::fromLocal8Bit( qgetenv( "GISDBASE" ) );
563 const QString loc = QString::fromLocal8Bit( qgetenv( "LOCATION_NAME" ) );
564 const QString ms = mapset.isEmpty()
565 ? QString::fromLocal8Bit( qgetenv( "MAPSET" ) )
566 : mapset;
567
568 if ( grassBin.isEmpty() )
569 {
570 QMessageBox::warning( this, "Build Overviews",
571 "Cannot find the 'grass' executable in PATH.\n"
572 "Make sure GRASS GIS is installed and /usr/local/bin is in PATH." );
573 return;
574 }
575 if ( gisdb.isEmpty() || loc.isEmpty() || ms.isEmpty() )
576 {
577 QMessageBox::warning( this, "Build Overviews",
578 "No active GRASS session (GISDBASE/LOCATION_NAME/MAPSET not set).\n"
579 "Open a mapset first." );
580 return;
581 }
582
583 // --- Get native resolution from the layer ---
584 double ewRes = 0, nsRes = 0;
585 int nRows = 0, nCols = 0;
586 for ( GrassMapLayer *l : mCanvas->layers() )
587 {
588 if ( l->name() == name )
589 {
590 auto *rl = dynamic_cast<GrassRasterMapLayer *>( l );
591 if ( rl ) { ewRes = rl->ewRes(); nsRes = rl->nsRes(); }
592 break;
593 }
594 }
595 if ( ewRes <= 0 || nsRes <= 0 )
596 {
597 QMessageBox::warning( this, "Build Overviews",
598 QString( "Cannot determine resolution for '%1'." ).arg( name ) );
599 return;
600 }
601
602 // --- Decide which levels to build ---
603 // Build as many levels as make sense (output still > 64 cells wide/tall)
604 const GrassExtent ext = [&]() -> GrassExtent {
605 for ( GrassMapLayer *l : mCanvas->layers() )
606 if ( l->name() == name ) return l->extent();
607 return {};
608 }();
609 nCols = ext.isValid() ? static_cast<int>( ext.width() / ewRes ) : 0;
610 nRows = ext.isValid() ? static_cast<int>( ext.height() / nsRes ) : 0;
611
612 QList<int> levels;
613 for ( int L : { 2, 4, 8, 16, 32, 64 } )
614 if ( nCols / L >= 64 && nRows / L >= 64 )
615 levels << L;
616
617 if ( levels.isEmpty() )
618 {
619 QMessageBox::information( this, "Build Overviews",
620 "Raster is too small to benefit from overview levels." );
621 return;
622 }
623
624 // --- Build the shell script (cascaded resample) ---
625 // Each level reads from the PREVIOUS level (not the original), so total
626 // I/O is O(1) × original size (geometric series: 1 + 1/4 + 1/16 + …).
627 const QString ref = name + "@" + ms;
628 QStringList cmds;
629 cmds << "set -e"; // abort on first error
630
631 QString prevMap = name + "@" + ms;
632 QStringList levelLines; // for the marker file
633
634 for ( int L : levels )
635 {
636 const QString ovrName = name + "_L" + QString::number( L );
637 const double ovrEwRes = ewRes * L;
638 const double ovrNsRes = nsRes * L;
639
640 // Set region to the source map's extent at the overview resolution
641 cmds << QString( "g.region raster=%1 ewres=%2 nsres=%3 --quiet" )
642 .arg( prevMap )
643 .arg( ovrEwRes, 0, 'g', 10 )
644 .arg( ovrNsRes, 0, 'g', 10 );
645
646 cmds << QString( "r.resamp.stats input=%1 output=%2@%3 method=average --overwrite --quiet" )
647 .arg( prevMap, ovrName, ms );
648
649 prevMap = ovrName + "@" + ms;
650 levelLines << QString( "%1 %2" ).arg( L ).arg( ovrName );
651 }
652
653 // Restore region to original raster
654 cmds << "g.region raster=" + ref + " --quiet";
655
656 const QString script = cmds.join( "\n" );
657
658 // --- Progress dialog ---
659 QDialog dlg( this );
660 dlg.setWindowTitle( QString( "Building overviews for %1" ).arg( name ) );
661 dlg.resize( 600, 400 );
662
663 auto *vbox = new QVBoxLayout( &dlg );
664 auto *status = new QLabel( "Starting…", &dlg );
665 auto *log = new QPlainTextEdit( &dlg );
666 log->setReadOnly( true );
667 log->setFont( QFont( "monospace", 9 ) );
668 auto *btns = new QDialogButtonBox( QDialogButtonBox::Cancel, &dlg );
669 vbox->addWidget( status );
670 vbox->addWidget( log, 1 );
671 vbox->addWidget( btns );
672
673 QProcess *proc = new QProcess( &dlg );
674 proc->setProcessChannelMode( QProcess::MergedChannels );
675 proc->setProcessEnvironment( GrassEnv::instance()->buildSubprocessEnv() );
676
677 connect( proc, &QProcess::readyRead, &dlg, [proc, log] {
678 log->appendPlainText( QString::fromLocal8Bit(
679 proc->readAll() ).trimmed() );
680 } );
681
682 bool succeeded = false;
683 connect( proc,
684 QOverload<int, QProcess::ExitStatus>::of( &QProcess::finished ),
685 &dlg, [&]( int code, QProcess::ExitStatus ) {
686 succeeded = ( code == 0 );
687 status->setText( succeeded ? "Done." : "Failed — see log above." );
688 btns->setStandardButtons( QDialogButtonBox::Ok );
689 } );
690
691 connect( btns, &QDialogButtonBox::rejected, &dlg, [&] {
692 proc->kill();
693 dlg.reject();
694 } );
695 connect( btns, &QDialogButtonBox::accepted, &dlg, &QDialog::accept );

697 // Run: grass /gisdb/loc/ms --exec bash -c "script"
