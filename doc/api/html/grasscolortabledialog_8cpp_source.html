161void GrassColorTableDialog::showStoredColorTable()
162{
163 const QByteArray nameB = mMapName.toLocal8Bit();
164 const QByteArray mapsetB = mMapset.toLocal8Bit();
165 const char *cName = nameB.constData();
166 const char *cMapset = mapsetB.isEmpty() ? "" : mapsetB.constData();
167
168 struct Colors colors;
169 if ( Rast_read_colors( cName, cMapset, &colors ) < 0 )
170 return; // no stored table — keep the file-based preview from loadGradient
171
172 // Read the raster value range
173 struct FPRange fpRange;
174 Rast_init_fp_range( &fpRange );
175 Rast_read_fp_range( cName, cMapset, &fpRange );
176 DCELL dmin = 0.0, dmax = 1.0;
177 Rast_get_fp_range_min_max( &fpRange, &dmin, &dmax );
178 if ( dmin == dmax ) { dmax = dmin + 1.0; }
179
180 // Sample 128 points across the range → gradient stops (0–1)
181 const int nSamples = 128;
182 QVector<QPair<double, QColor>> stops;
183 stops.reserve( nSamples + 1 );
184 for ( int i = 0; i <= nSamples; ++i )
185 {
186 const DCELL v = dmin + static_cast<double>(i) / nSamples * (dmax - dmin);
187 int r, g, b;
188 Rast_get_d_color( &v, &r, &g, &b, &colors );
189 stops.append( { static_cast<double>(i) / nSamples, QColor(r, g, b) } );
190 }
191 Rast_free_colors( &colors );
192
193 mBar->setGradient( stops );
194 mStatus->setText( "Showing stored colour table from GRASS database." );
195}

205static QColor parseGrassNamedColor( const QString &s )
206{
207 // First try Qt's built-in colour names (covers "red","blue","cyan","yellow",…)
208 QColor c( s.toLower() );
209 if ( c.isValid() ) return c;
210 // Extra GRASS aliases not in Qt
211 static const QHash<QString,QColor> extra = {
212 {"grey", {128,128,128}},
213 {"gray", {128,128,128}},
214 {"violet", {238,130,238}},
215 {"aqua", {0,255,255}},
216 {"brown", {139,69,19}},
217 {"orange", {255,165,0}},
218 };
219 return extra.value( s.toLower(), Qt::gray );
220}

222// ---------------------------------------------------------------------------
223// loadGrassColorFile — read $GISBASE/etc/colors/<name> and return stops [0,1]
224// ---------------------------------------------------------------------------
225using ColorStops = QVector<QPair<double, QColor>>;
226
227static ColorStops loadGrassColorFile( const QString &gisBase,
228 const QString &tableName )
229{
230 QFile f( gisBase + "/etc/colors/" + tableName );
231 if ( !f.open( QIODevice::ReadOnly | QIODevice::Text ) )
232 return {};
233
234 struct Raw { double val; QColor color; bool isPct; };
235 QVector<Raw> rows;
236 QTextStream ts( &f );
237 while ( !ts.atEnd() )
238 {
239 const QString line = ts.readLine().trimmed();
240 if ( line.isEmpty() || line.startsWith('#') ) continue;
241
242 const QStringList tok = line.split( QRegularExpression("\\s+"),
243 Qt::SkipEmptyParts );
244 if ( tok.size() < 2 ) continue;
245
246 const bool isPct = tok[0].endsWith('%');
247 const QString vs = isPct ? tok[0].chopped(1) : tok[0];
248 bool ok;
249 double val = vs.toDouble( &ok );
250 if ( !ok ) continue; // skip "nv", "default", text tokens
251
252 QColor col;
253 if ( tok.size() >= 4 ) // R G B triplet
254 col = { tok[1].toInt(), tok[2].toInt(), tok[3].toInt() };
255 else if ( tok[1].contains(':') ) // R:G:B
256 {
257 const QStringList rgb = tok[1].split(':');
258 if ( rgb.size() == 3 )
259 col = { rgb[0].toInt(), rgb[1].toInt(), rgb[2].toInt() };
260 }
261 else // colour name
262 col = parseGrassNamedColor( tok[1] );
263
264 if ( col.isValid() )
265 rows.append( { val, col, isPct } );
266 }
267
268 if ( rows.isEmpty() ) return {};
269
270 // If all stops are percentages, normalise directly
271 const bool allPct = std::all_of( rows.begin(), rows.end(),
272 []( const Raw &r ){ return r.isPct; } );
273 if ( allPct )
274 {
275 ColorStops out;
276 for ( const Raw &r : rows )
277 out.append( { r.val / 100.0, r.color } );
278 return out;
279 }
280
281 // Absolute values — normalise by min/max
282 double lo = rows[0].val, hi = rows[0].val;
283 for ( const Raw &r : rows ) { lo = std::min(lo,r.val); hi = std::max(hi,r.val); }
284 const double span = hi - lo;
285 if ( span == 0.0 ) return { { 0.0, rows[0].color }, { 1.0, rows[0].color } };
286
287 ColorStops out;
288 for ( const Raw &r : rows )
289 out.append( { (r.val - lo) / span, r.color } );
290 return out;
291}

296void GrassColorTableDialog::loadGradient( const QString &colorTable )
297{
298 const QString gisBase = QString::fromLocal8Bit( qgetenv( "GISBASE" ) );
299
300 // 1. Try reading the file from $GISBASE/etc/colors/
301 if ( !gisBase.isEmpty() )
302 {
303 const ColorStops stops = loadGrassColorFile( gisBase, colorTable );
304 if ( !stops.isEmpty() ) { mBar->setGradient( stops ); return; }
305 }
306
307 // 2. Fallback for computed tables that have no file (grey.eq, grey.log, …)
308 static const QHash<QString, ColorStops> kFallback = {
309 { "grey.eq", { {0.0, Qt::black}, {1.0, Qt::white} } },
310 { "grey.log", { {0.0, Qt::black}, {1.0, Qt::white} } },
311 { "corine", { {0.0,{230,0,77}},{0.3,{255,255,168}},{0.6,{0,100,0}},{0.9,{0,128,255}},{1.0,{255,255,255}} } },
312 };
313 auto it = kFallback.find( colorTable );
314 if ( it != kFallback.end() ) { mBar->setGradient( it.value() ); return; }
315
316 // 3. Last resort: black→white generic gradient
317 mBar->setGradient( { {0.0, Qt::black}, {1.0, Qt::white} } );
318}

366bool GrassColorTableDialog::runRColors( const QString &colorTable )
367{
368 const QString gisBase = QString::fromLocal8Bit( qgetenv( "GISBASE" ) );
369 if ( gisBase.isEmpty() ) { mStatus->setText( "GISBASE not set." ); return false; }
370
371 const QString mapRef = mMapset.isEmpty()
372 ? mMapName : mMapName + "@" + mMapset;
373
374 QStringList args = { "map=" + mapRef, "color=" + colorTable };
375 if ( mInvertCheck ->isChecked() ) args << "-n";
376 if ( mEqualizeCheck->isChecked() ) args << "-e";
377
378 // Persist chosen table so the dialog reopens on the correct selection
379 QSettings( "qtgrass", "qtgrass" ).setValue( "colortable/" + mapRef, colorTable );
380
381 QProcess proc;
382 proc.setProcessEnvironment( GrassEnv::instance()->buildSubprocessEnv() );
383 proc.start( gisBase + "/bin/r.colors", args );
384 proc.waitForFinished( 10000 );
385
386 if ( proc.exitCode() != 0 )
387 {
388 mStatus->setStyleSheet( "color:#c00;" );
389 mStatus->setText( "r.colors failed: " +
390 proc.readAllStandardError().simplified() );
391 return false;
392 }
393 return true;
