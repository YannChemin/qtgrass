41class DeleteLineCmd : public QUndoCommand
42{
43 public:
44 DeleteLineCmd( Map_info *map, long long lineId,
45 line_pnts *pnts, line_cats *cats )
46 : QUndoCommand( "Delete feature" )
47 , mMap( map ), mLineId( lineId )
48 {
49 // Capture the geometry for undo
50 mPntsCopy = Vect_new_line_struct();
51 mCatsCopy = Vect_new_cats_struct();
52 // (The caller already read the geometry before deleting)
53 mType = Vect_read_line( mMap, mPntsCopy, mCatsCopy,
54 static_cast<int>( lineId ) );
55 Q_UNUSED( pnts ); Q_UNUSED( cats );
56 }
57
58 ~DeleteLineCmd() override
59 {
60 if ( mPntsCopy ) Vect_destroy_line_struct( mPntsCopy );
61 if ( mCatsCopy ) Vect_destroy_cats_struct( mCatsCopy );
62 }
63
64 void undo() override
65 {
66 // Re-add the deleted feature (gets a new internal ID)
67 if ( mType > 0 && mPntsCopy )
68 Vect_write_line( mMap, mType, mPntsCopy, mCatsCopy );
69 }
70
71 void redo() override
72 {
73 // Re-delete it — but after undo the line has a new ID, so just skip
74 // (full redo tracking requires tracking updated offsets)
75 }
76
77 private:
78 Map_info *mMap = nullptr;
79 long long mLineId = 0;
80 int mType = -1;
81 line_pnts *mPntsCopy = nullptr;
82 line_cats *mCatsCopy = nullptr;
83};

102bool GrassVectorEditor::open( const QString &name, const QString &mapset )
103{
104 if ( mOpen ) close();
105
106 mMap = new Map_info();
107 const QByteArray nb = name.toLocal8Bit();
108 const QByteArray mb = mapset.toLocal8Bit();
109
110 // Vect_open_update() calls G_fatal_error (→ abort) when the map does not
111 // exist. Check first with G_find_vector2(), and create the map via
112 // Vect_open_new() if it is missing so the user can start a fresh digitise
113 // session without pre-existing data.
114 const char *msArg = mb.isEmpty() ? "" : mb.constData();
115 const char *foundMs = G_find_vector2( nb.constData(), msArg );
116
117 int openRet = -1;
118 if ( foundMs )
119 {
120 // Existing map → open for update in the mapset where it was found
121 openRet = Vect_open_update( mMap, nb.constData(), foundMs );
122 if ( openRet < 0 )
123 {
124 mError = QString( "Cannot open '%1@%2' for editing." )
125 .arg( name ).arg( QString::fromUtf8( foundMs ) );
126 qCWarning(lcVectorEd) << "GrassVectorEditor:" << mError;
127 delete mMap; mMap = nullptr;
128 return false;
129 }
130 }
131 else
132 {
133 // Map does not exist → create it in the current (or requested) mapset.
134 // Vect_open_new() opens the new map for writing; topology is built
135 // in close() via Vect_build().
136 openRet = Vect_open_new( mMap, nb.constData(), WITHOUT_Z );
137 if ( openRet < 0 )
138 {
139 mError = QString( "Cannot create new vector map '%1'." ).arg( name );
140 qCWarning(lcVectorEd) << "GrassVectorEditor:" << mError;
141 delete mMap; mMap = nullptr;
142 return false;
143 }
144 qCInfo(lcVectorEd) << "GrassVectorEditor: created new vector map" << name;
145 }
146
147 mPnts = Vect_new_line_struct();
148 mCats = Vect_new_cats_struct();
149 mName = name;
150 mMapset = mapset.isEmpty() ? QString::fromUtf8( G_mapset() ) : mapset;
151 mOpen = true;
152 mUndoStack->clear();
153 mNextCat = static_cast<int>( Vect_get_num_lines( mMap ) ) + 1;
154
155 emit sessionOpened( name );
156 qCInfo(lcVectorEd) << "GrassVectorEditor: opened" << name << "for editing";
157 return true;
158}

161{
162 if ( !mOpen ) return true;

166

169

193// -----------------------------------------------------------------------
194long long GrassVectorEditor::writeAndRecord( int vtype,
195 line_pnts *pnts,
196 line_cats *cats )
197{
198 const off_t id = Vect_write_line( mMap, vtype, pnts, cats );
199 if ( id < 0 ) return -1;
200
201 // Track for undo
202 const QString desc = ( vtype == GV_POINT ) ? "Add point" :
203 ( vtype == GV_LINE ) ? "Add line" :
204 ( vtype == GV_BOUNDARY ) ? "Add boundary" : "Add feature";
205 mUndoStack->push( new AddLineCmd( mMap, static_cast<long long>( id ), desc ) );
206
207 emit geometryChanged();
208 return static_cast<long long>( id );

232long long GrassVectorEditor::addLine( const QVector<QPointF> &points )
233{
234 if ( !mOpen || points.size() < 2 ) return -1;
235
236 Vect_reset_line( mPnts );
237 for ( const QPointF &p : points )
238 {
239 double x = p.x(), y = p.y();
240 // Snap only the first and last vertex
241 if ( &p == &points.first() || &p == &points.last() )
242 snapPoint( x, y );
243 Vect_append_point( mPnts, x, y, 0.0 );
244 }
245 Vect_reset_cats( mCats );
246 Vect_cat_set( mCats, mLayer, mNextCat++ );
247
248 return writeAndRecord( GV_LINE, mPnts, mCats );

251// -----------------------------------------------------------------------
252// addArea — boundary + centroid
253// -----------------------------------------------------------------------
254long long GrassVectorEditor::addArea( const QVector<QPointF> &points )
255{
256 if ( !mOpen || points.size() < 3 ) return -1;
257
258 Vect_reset_line( mPnts );
259 for ( const QPointF &p : points )
260 Vect_append_point( mPnts, p.x(), p.y(), 0.0 );
261
262 // Close the boundary if not already closed
263 if ( points.first() != points.last() )
264 Vect_append_point( mPnts, points.first().x(), points.first().y(), 0.0 );
265
266 Vect_reset_cats( mCats );
267 Vect_cat_set( mCats, mLayer, mNextCat );
268
269 const long long boundId = writeAndRecord( GV_BOUNDARY, mPnts, mCats );
270 if ( boundId < 0 ) return -1;
271
272 // Add centroid at the polygon centroid (simple bbox center for now)
273 double cx = 0.0, cy = 0.0;
274 for ( const QPointF &p : points ) { cx += p.x(); cy += p.y(); }
275 cx /= points.size(); cy /= points.size();
276
277 Vect_reset_line( mPnts );
278 Vect_append_point( mPnts, cx, cy, 0.0 );
279 Vect_reset_cats( mCats );
280 Vect_cat_set( mCats, mLayer, mNextCat++ );
281
282 writeAndRecord( GV_CENTROID, mPnts, mCats );
283 return boundId;
284}

365// nearestLine — proximity query using Vect_find_line
366// -----------------------------------------------------------------------
367long long GrassVectorEditor::nearestLine( double x, double y,
368 double maxDist, int type ) const
369{
370 if ( !mOpen || !mMap ) return -1;
371
372 // Vect_find_line searches the spatial index for the nearest feature.
373 // type: bitmask of GV_POINT | GV_LINE | GV_BOUNDARY | GV_CENTROID etc.
374 // Pass -1 (all bits set) to search every type.
375 const int ftype = ( type < 0 )
376 ? ( GV_POINT | GV_LINE | GV_BOUNDARY | GV_CENTROID | GV_FACE | GV_KERNEL )
377 : type;
378
379 const int id = Vect_find_line( mMap, x, y, 0.0, ftype,
380 maxDist, 0 /*not 3D*/, 0 /*with_z*/ );
381 return ( id > 0 ) ? static_cast<long long>( id ) : -1LL;
382}
